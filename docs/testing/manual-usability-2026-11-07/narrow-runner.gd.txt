extends SceneTree
const Manual=preload("res://tests/manual_combo_usability_tests.gd")
const Main=preload("res://game/main.gd")
var passed:=0
var failed:=0
func _initialize()->void:_run.call_deferred()
func _run()->void:
 var game=Main.new()
 game.persistence_enabled=false
 root.add_child(game)
 await physics_frame
 await process_frame
 await Manual.run(self,game,_check)
 game.queue_free()
 await process_frame
 print("MANUAL NARROW: ",passed," passed, ",failed," failed")
 quit(1 if failed else 0)
func _check(ok:bool,label:String)->void:
 if ok:passed+=1
 else:failed+=1;print("FAIL: ",label)
