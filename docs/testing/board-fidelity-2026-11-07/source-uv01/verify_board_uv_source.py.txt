 (passes if ok else failures).append(label)
def protected(op,oa,oi,np_,na,ni):return op==np_ and np.array_equal(oi,ni) and all(oa[k].tobytes()==na[k].tobytes() for k in oa)
def triangles(a,ix):return a['POSITION'][ix].reshape(-1,3,3).astype(np.float64)
def geometry(a,ix):
 p=triangles(a,ix);cross=np.cross(p[:,1]-p[:,0],p[:,2]-p[:,0]);twicearea=np.sqrt(np.sum(cross*cross,axis=1));return cross/twicearea[:,None],twicearea

def uv_jac(a,ix):
 uv=a['TEXCOORD_0'][ix].reshape(-1,3,2).astype(np.float64);e=uv[:,1]-uv[:,0];f=uv[:,2]-uv[:,0];return e[:,0]*f[:,1]-e[:,1]*f[:,0]
def tangent_valid(a):
 t=a['TANGENT'];n=a['NORMAL'];return bool(np.isfinite(t).all() and np.max(np.abs(np.sqrt(np.sum(t[:,:3]**2,axis=1))-1))<2e-6 and np.max(np.abs(np.sum(t[:,:3]*n,axis=1)))<2e-6 and np.all(np.abs(t[:,3])==1))
