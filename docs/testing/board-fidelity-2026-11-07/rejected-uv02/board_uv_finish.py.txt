"""Private offline indexed-GLB wood-UV finish; no native scene/runtime mutations.

def sha(b):return hashlib.sha256(b).hexdigest()

def face_data(vertices,indices,uv):
 p=vertices[indices].astype(np.float64).reshape(-1,3,3);t=uv[indices].astype(np.float64).reshape(-1,3,2)
 cross=np.cross(p[:,1]-p[:,0],p[:,2]-p[:,0]);length=np.linalg.norm(cross,axis=1)
 normal=cross/length[:,None];du=t[:,1]-t[:,0];dv=t[:,2]-t[:,0];jac=du[:,0]*dv[:,1]-du[:,1]*dv[:,0]

def perimeter_parameters(points):

 # Begin at one exact finite convex-hull vertex, retaining seam provenance.
 begin=int(np.lexsort((outline[:,0],outline[:,1]))[0]);outline=np.roll(outline,-begin,axis=0)
 edge=np.roll(outline,-1,axis=0)-outline;lens=np.linalg.norm(edge,axis=1);cumulative=np.r_[0,np.cumsum(lens)];total=float(cumulative[-1])
 result=[]
 for at in xz:
  t=np.clip(np.sum((at-outline)*edge,axis=1)/lens**2,0,1)
  distances=np.sum((outline+edge*t[:,None]-at)**2,axis=1);nearest=int(np.argmin(distances))
  result.append((cumulative[nearest]+t[nearest]*lens[nearest])/total)
 return np.array(result),total,outline
