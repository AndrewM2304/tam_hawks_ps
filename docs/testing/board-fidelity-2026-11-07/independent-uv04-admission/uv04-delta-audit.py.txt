from pathlib import Path
import ast,base64,collections,hashlib,io,json,struct
import numpy as np
from PIL import Image
R=Path('/workspace/tam_hawks_ps');checks=[]
def ck(label,v):checks.append({'label':label,'pass':bool(v)})
tree=ast.parse((R/'.local/qa-fidelity/review_board_uv03.py').read_text())
cls=next(n for n in tree.body if isinstance(n,ast.ClassDef) and n.name=='GLB')
ns={'struct':struct,'json':json,'np':np};exec(compile(ast.Module(body=[cls],type_ignores=[]),'<independentGLBreader>','exec'),ns)
old=ns['GLB'](R/'.local/board-fidelity/candidate-uv03/tam_skateboard.glb')
new=ns['GLB'](R/'.local/board-fidelity/candidate-uv04/tam_skateboard.glb')
for i in range(4):
    po,ao,io_=old.primitive(i);pn,an,inn=new.primitive(i)
    ck('UV03→UV04 primitive'+str(i)+' every attribute and index exact',ao.keys()==an.keys() and all(np.array_equal(v,an[k]) for k,v in ao.items()) and np.array_equal(io_,inn))
for key in ['nodes','scenes','scene','materials','textures','samplers']:
    ck('UV03→UV04 scene contract '+key+' exact',old.j.get(key)==new.j.get(key))
a=Image.open(io.BytesIO(old.image()));b=Image.open(io.BytesIO(new.image()))
ck('only unused source PNG alpha channel removed RGBA→RGB',a.mode=='RGBA' and b.mode=='RGB' and a.size==b.size and a.getextrema()[-1]==(255,255))
ck('all decoded atlas RGBA pixels exact',a.convert('RGBA').tobytes()==b.convert('RGBA').tobytes())
native=json.loads((R/'.local/board-fidelity/native-component-08/native-result.json').read_text())['records']
def arr(x,d,w):return np.frombuffer(base64.b64decode(x['base64']),dtype=d).reshape(-1,w)
def tris(s):
    p=arr(s['positions'],'<f4',3);n=arr(s['normals'],'<f4',3);i=arr(s['indices'],'<i4',1).ravel()
    v=np.c_[p,n][i].reshape(-1,3,6)
    result=[]
    for tri in v:
        q=tuple(tuple(float(z)for z in row)for row in tri)
        result.append(min(q,q[1:]+q[:1],q[2:]+q[:2]))
    return collections.Counter(result)
for i,(a,b) in enumerate(zip(native[0]['surfaces'],native[1]['surfaces'])):
    ck('native08 surface'+str(i)+' exact winding-aware POSITION+NORMAL triangle multiset',tris(a)==tris(b))

No importer/raster/LOD/motion/contact equivalence inferred.','inputs':{'uv03':hashlib.sha256(old.data).hexdigest(),'uv04':hashlib.sha256(new.data).hexdigest(),'native08':hashlib.sha256((R/'.local/board-fidelity/native-component-08/native-result.json').read_bytes()).hexdigest()}}
o=R/'docs/testing/board-fidelity-2026-11-07/independent-uv04-admission';(o/'uv04-delta-supplement.json').write_text(json.dumps(report,indent=2)+'\n');(o/'uv04-delta-audit.py.txt').write_text(Path(__file__).read_text())
