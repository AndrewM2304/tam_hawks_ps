   if textures.has(texture.get_instance_id()):continue
   var image:Image=texture.get_image()
   if image==null:failures+=1;push_error("Bound Texture2D.get_image unavailable");return {"label":label,"available":false}
   var flat:Image=image.duplicate() as Image
   flat.clear_mipmaps()
   var path:String="res://frames/textures/"+label+".png"
   if flat.save_png(path)!=OK:failures+=1;push_error("Bound texture readback PNG write failed")
   textures[texture.get_instance_id()]={"path":texture.resource_path,"native_image_format":image.get_format(),"mipmap_count":image.get_mipmap_count(),"data_bytes":image.get_data().size(),"all_levels_sha256":_rgba_sha(image),"base_level_sha256":_rgba_sha(flat),"png_path":path,"png_sha256":FileAccess.get_sha256(path),"scope":"actual bound Texture2D.get_image API in graphical context; separate from viewport draw pixels; backend readback origin not independently audited"}

func _marker_pixels(image:Image,clean_image:Image,projected:Vector2,channel:int)->Dictionary:
 var total:=Vector2.ZERO
 var count:int=0
 for y:int in range(maxi(0,int(projected.y)-12),mini(image.get_height(),int(projected.y)+13)):
  for x:int in range(maxi(0,int(projected.x)-12),mini(image.get_width(),int(projected.x)+13)):
   var pixel:Color=image.get_pixel(x,y)
   var clean:Color=clean_image.get_pixel(x,y)

func _pack(value:Variant)->Variant:
 if value is Vector3:return [float(value.x),float(value.y),float(value.z)]
 if value is Vector2 or value is Vector2i:return [float(value.x),float(value.y)]
 if value is Vector4:return [float(value.x),float(value.y),float(value.z),float(value.w)]
 if value is Projection:return [_pack(value.x),_pack(value.y),_pack(value.z),_pack(value.w)]
 if value is Basis:return [_pack(value.x),_pack(value.y),_pack(value.z)]
 if value is Transform3D:return {"origin":_pack(value.origin),"basis":_pack(value.basis)}
 if value is Dictionary:
  var result:Dictionary={}
  for key:Variant in value:result[str(key)]=_pack(value[key])
  return result
 if value is Array:
  var result:Array=[]
  for item:Variant in value:result.append(_pack(item))
  return result
 return value
