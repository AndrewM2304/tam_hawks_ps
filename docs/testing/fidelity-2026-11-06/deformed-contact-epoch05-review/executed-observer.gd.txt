class WrongNativeFace extends Candidate:
	var foreign_rid: RID
	var wrong_shape: int=-1
	var omit_shape:=false
	var altered:=false
	func _face(ray: PhysicsRayQueryParameters3D,from: Vector3,to: Vector3) -> Dictionary:
		var actual: Dictionary=super._face(ray,from,to)
		if not actual.is_empty():
			if foreign_rid.is_valid():actual.rid=foreign_rid
			if wrong_shape>=0:actual.shape=wrong_shape
			if omit_shape:actual.erase("shape")
			altered=true
		return actual

func _native_face_guards() -> void:
	var f:=Arm._fixture(self,true)
	await Arm._approach(self,f)

	await Wall._frames(self,6)
	Wall._release()
	var rider: CharacterBody3D=f.rider
	rider.set_physics_process(false)
	# A second real native shape exists on this same body, well outside the
	# fixture. Negative ray metadata deliberately substitutes its valid index.
	var second:=CollisionShape3D.new()
	var box:=BoxShape3D.new();box.size=Vector3(.1,.1,.1)
	second.shape=box;second.position=Vector3(15,15,15)
	f.gate._panel.add_child(second)
	for mode: int in 3:
		var wrong:=WrongNativeFace.new();wrong.prepare(rider)
		if mode==0:wrong.wrong_shape=1
		elif mode==1:wrong.foreign_rid=f.root.get_child(0).get_rid()
		else:wrong.omit_shape=true
		rider._obstacle_arm_fit=wrong
		var pose:=Arm._arm_pose(rider)
		var query:=Arm._query_state(rider)
		wrong.apply(1.0/60.0)
		_check(wrong.altered and wrong.accepted==0 and Arm._arm_pose(rider)==pose and Arm._query_state(rider)==query and wrong.maximum_calls<=12,
			"native face: "+["same body different real shape","different real body RID","missing integer shape"][mode]+" cannot authorize an arm correction")
	f.viewport.queue_free()
	await process_frame

func _points(helper: RefCounted,arm: Dictionary,overrides: Dictionary) -> PackedVector3Array:
	var result:=PackedVector3Array()
	var palette: Array[Transform3D]=[]
	for bind: int in helper.jacket.skin.get_bind_count():
		var bone: int=helper._bone(bind)
		var pose: Transform3D=overrides.get(bone,helper.skeleton.get_bone_global_pose(bone))
		palette.append(pose*helper.jacket.skin.get_bind_pose(bind))
	for vertex: int in arm.points.size():
		var x:=Vector3.ZERO;var y:=Vector3.ZERO;var z:=Vector3.ZERO;var origin:=Vector3.ZERO
		for index: int in range(arm.offsets[vertex],arm.offsets[vertex+1]):
			var transform: Transform3D=palette[arm.binds[index]]
			var weight: float=arm.weights[index]
			x+=transform.basis.x*weight;y+=transform.basis.y*weight;z+=transform.basis.z*weight;origin+=transform.origin*weight
		result.append(helper.jacket.global_transform*(Transform3D(Basis(x,y,z),origin)*Vector3(arm.points[vertex])))
	return result

func _wrong_world_weighted_points(helper: RefCounted,arm: Dictionary) -> PackedVector3Array:
	var result:=PackedVector3Array()
	for vertex: int in arm.points.size():
		var point:=Vector3.ZERO
		for index: int in range(arm.offsets[vertex],arm.offsets[vertex+1]):
			var bind: int=arm.binds[index]
			point+=(helper.jacket.global_transform*helper.skeleton.get_bone_global_pose(helper._bone(bind))*helper.jacket.skin.get_bind_pose(bind)*Vector3(arm.points[vertex]))*float(arm.weights[index])
		result.append(point)
	return result

func _check(ok: bool,label: String) -> void:
	if ok:passed+=1;print("PASS ",label)
	else:failed.append(label);print("FAIL ",label)
