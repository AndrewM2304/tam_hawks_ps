const MODES = ["before-a","native-default-bias","before-b"]
func _run() -> void:
 var ready: bool=await _start_actual_main()
 if not ready:return
 game.skater.process_mode=Node.PROCESS_MODE_DISABLED
 var sun: DirectionalLight3D=game.world.get_node("LateAfternoonSun")
 _baseline_bias=sun.shadow_bias
 _baseline_normal_bias=sun.shadow_normal_bias
 _check(is_equal_approx(_baseline_bias,.035) and is_equal_approx(_baseline_normal_bias,.55),"combined current-source comparison starts from the declared actual low-bias profile")
 var default_light:=DirectionalLight3D.new()
 _native_defaults={"shadow_bias":default_light.shadow_bias,"shadow_normal_bias":default_light.shadow_normal_bias,"source":"actual unadded DirectionalLight3D getters, probe freed before frames"}
 default_light.free()
 _lighting_before=_inventory()
 _check(_lighting_before.water.size()==3,"three installed exact shared-water instances remain resident")
 for water: Dictionary in _lighting_before.water:_check(water.cast_shadow==GeometryInstance3D.SHADOW_CASTING_SETTING_OFF,"installed water policy is constant through every lighting mode")
 var basin: MeshInstance3D=game.world.get_node("ExchangeFountain/FountainBasin")
 _check(basin.mesh is ArrayMesh and basin.mesh.surface_get_arrays(0)[Mesh.ARRAY_INDEX].size()/3==512 and basin.cast_shadow==GeometryInstance3D.SHADOW_CASTING_SETTING_ON,"current exact512closed basin is integrated and remains a real caster")
 _create_pixel_proofs()
 for mode_index: int in MODES.size():
  var mode: String=MODES[mode_index]
  sun.shadow_bias=float(_native_defaults.shadow_bias) if mode=="native-default-bias" else _baseline_bias
  sun.shadow_normal_bias=float(_native_defaults.shadow_normal_bias) if mode=="native-default-bias" else _baseline_normal_bias
  var inventory:=_inventory()
  var canonical: Dictionary=inventory.duplicate(true)
  canonical.sun.bias=_lighting_before.sun.bias
  canonical.sun.normal_bias=_lighting_before.sun.normal_bias
  _check(canonical==_lighting_before,"only two declared sun-bias values change; every native caster/receiver/geometry/material/map/collider/sky state remains")
  _lighting_modes[mode]=inventory
  var view_index:=0
  for view: Dictionary in _lighting_views():
   var camera: Camera3D
   if view.id=="ordinary-chase":
    camera=game.skater._camera
    game.hud.visible=true
   else:
    camera=_inspection
    camera.global_position=view.at
    camera.look_at(view.target)
    game.hud.visible=false
   camera.current=true
   _active_proof_camera=camera
   _position_proofs(mode_index,view_index)
   await process_frame
   await create_timer(0.0,true).timeout
   var physics_before:=Engine.get_physics_frames()
   var process_before:=Engine.get_process_frames()
   var image: Image=await _render_current_frame()
   var proofs:=_verify_pixel_proofs(image)
   var projection: Projection=camera.get_camera_projection()
   var filename: String=mode+"-"+str(view.id)+".png"
   var stored:=_store_frame(image,filename)
   _lighting_frames.append({"file":filename,"mode":mode,"view":view.id,"pixels":stored,"physics_before":physics_before,"physics_after":Engine.get_physics_frames(),"process_before":process_before,"process_after":Engine.get_process_frames(),"pixel_proofs":proofs,"camera_basis":[_v(camera.global_basis.x),_v(camera.global_basis.y),_v(camera.global_basis.z)],"projection_columns":[_v4(projection.x),_v4(projection.y),_v4(projection.z),_v4(projection.w)],"keep_aspect":camera.keep_aspect,"near":camera.near,"far":camera.far,"viewport_size":[root.size.x,root.size.y],"camera_position":_v(camera.global_position),"camera_forward":_v(-camera.global_basis.z),"fov":camera.fov,"hud_visible":game.hud.visible,"counters":_native_counters()})
   _write_lighting(false)
   view_index+=1
 _check(_lighting_frames.size()==18,"all18full-resolution six-view A/B/A light-contact originals complete")
 _write_lighting(true)
 print("ACTUAL_MAIN_LIGHTING_RESULT checks=",checks," failures=",failures)
 quit(1 if failures else 0)
