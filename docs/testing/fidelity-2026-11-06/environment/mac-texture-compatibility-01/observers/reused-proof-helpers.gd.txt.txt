func _render_current_frame() -> Image:
	# Let the engine flush current Node3D, Skeleton and Canvas notifications in
	# its normal render pass. Manual force_draw immediately after changing the
	# camera can read the previous camera, even when metadata is already current.
	var native_tick := Engine.get_physics_frames()
	var process_tick := Engine.get_process_frames()
	var started := Time.get_ticks_usec()
	RenderingServer.set_render_loop_enabled(true)
	await RenderingServer.frame_post_draw
	RenderingServer.set_render_loop_enabled(false)
	_check(Engine.get_physics_frames() == native_tick and Engine.get_process_frames() == process_tick,
		"natural readback completes the current frame without another physics/process tick")
	var image := root.get_texture().get_image()
	image.convert(Image.FORMAT_RGBA8)
	_check(image.get_width() == _native_size.x and image.get_height() == _native_size.y,
		"native capture retains every pixel of its declared "+str(_native_size)+" viewport")
	image.set_meta("draw_and_read_microseconds", Time.get_ticks_usec() - started)
	return image

func _store_frame(image: Image, relative: String) -> Dictionary:
	var started := Time.get_ticks_usec()
	var bytes := image.get_data()
	var hash := HashingContext.new()
	hash.start(HashingContext.HASH_SHA256)
	hash.update(bytes)
	var pixel_hash := hash.finish().hex_encode()
	if _raw_spool:
		var raw_path := output.path_join(relative.get_basename() + ".rgba")
		var file := FileAccess.open(raw_path + ".tmp", FileAccess.WRITE)
		_check(file != null, "raw pixel spool opened " + relative)
		if file != null:
			file.store_buffer(bytes)
			file.close()
			_check(DirAccess.rename_absolute(raw_path + ".tmp", raw_path) == OK,
				"complete native pixel spool published atomically " + relative)
	else:
		_check(image.save_png(output.path_join(relative)) == OK, "lossless frame saved " + relative)
	return {"format":"RGBA8", "rgba8_sha256":pixel_hash, "bytes":bytes.size(),
		"draw_and_read_microseconds":image.get_meta("draw_and_read_microseconds", 0),
		"hash_and_store_microseconds":Time.get_ticks_usec() - started,
		"encoding":"atomic raw spool to standard-library lossless PNG" if _raw_spool else "Godot lossless PNG"}

func _v(point: Vector3) -> Array:
	return [point.x, point.y, point.z]

func _check(ok: bool, label: String) -> void:
	checks += 1
	if not ok:
		failures += 1
		push_error("FAIL: " + label)

func _create_pixel_proofs() -> void:
	for color: Color in [Color.RED,Color.GREEN]:
		var mesh := MeshInstance3D.new()
		var sphere := SphereMesh.new()
		sphere.radius = 0.055
		sphere.height = 0.11
		sphere.radial_segments = 12
		sphere.rings = 6
		mesh.mesh = sphere
		var material := StandardMaterial3D.new()
		material.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED
		material.albedo_color = color
		material.no_depth_test = true
		mesh.material_override = material
		mesh.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
		root.add_child(mesh)
		_proof_markers.append(mesh)
	var canvas := CanvasLayer.new()
	root.add_child(canvas)
	_proof_swatch = ColorRect.new()
	_proof_swatch.position = Vector2(4,4)
	_proof_swatch.size = Vector2(18,18)
	canvas.add_child(_proof_swatch)

func _verify_pixel_proofs(image: Image) -> Dictionary:
	var records: Array[Dictionary] = []
	for index: int in _proof_markers.size():
		var point := _proof_markers[index].global_position
		var projected := _active_proof_camera.unproject_position(point)
		var center := Vector2.ZERO
		var count := 0
		for y: int in range(maxi(0,int(projected.y)-10),mini(image.get_height(),int(projected.y)+11)):
			for x: int in range(maxi(0,int(projected.x)-10),mini(image.get_width(),int(projected.x)+11)):
				var color := image.get_pixel(x,y)
				var found := color.r>0.75 and color.r-color.g>0.20 and color.r-color.b>0.20 if index==0 else color.g>0.75 and color.g-color.r>0.20 and color.g-color.b>0.20
				if found: center+=Vector2(x+0.5,y+0.5);count+=1
		if count: center/=float(count)
		var error := center.distance_to(projected)
		_check(count>=8 and error<=2.0,"projected independent3D marker matches actual pixels")
		records.append({"color":"red" if index==0 else "green","world_position":_v(point),"projected_xy":[projected.x,projected.y],"pixel_count":count,"centroid_xy":[center.x,center.y],"error_pixels":error})
	var expected := _proof_swatch.color
	var actual := image.get_pixel(10,10)
	var swatch_error := maxf(absf(expected.r-actual.r),maxf(absf(expected.g-actual.g),absf(expected.b-actual.b)))
	_check(swatch_error<=2.0/255.0,"current Canvas material/view swatch agrees with native pixels")
	return {"markers":records,"canvas_expected_rgb":[expected.r,expected.g,expected.b],"canvas_actual_rgb":[actual.r,actual.g,actual.b],"canvas_error":swatch_error,"canvas_sample_xy":[10,10]}

func _v4(value: Vector4) -> Array:
	return [value.x,value.y,value.z,value.w]
