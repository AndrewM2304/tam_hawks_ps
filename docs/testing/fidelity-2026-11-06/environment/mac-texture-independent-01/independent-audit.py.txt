def mip_bytes(w,h,block):
 total=0
 while True:
  total+=((w+3)//4)*((h+3)//4)*block
  if w==h==1:return total
  w=max(1,w//2);h=max(1,h//2)

def decode(folder,meta):
 data=(B/folder/meta['path']).read_bytes();check('compressed RGBA file hash '+folder+'/'+meta['path'],H(data)==meta['zstd_sha256'])
 raw=zstandard.ZstdDecompressor().decompress(data)
 check('decoded native RGBA identity '+folder+'/'+meta['path'],H(raw)==meta['rgba_sha256'] and len(raw)==meta['rgba_bytes']==meta['width']*meta['height']*4)
 return np.frombuffer(raw,np.uint8).reshape(meta['height'],meta['width'],4)

def normal(rgb):
 xy=rgb[...,:2].astype(np.float64)/127.5-1
 out=np.concatenate([xy,np.sqrt(np.maximum(0,1-np.sum(xy*xy,axis=-1)))[...,None]],axis=-1)
 return out/np.maximum(np.linalg.norm(out,axis=-1,keepdims=True),1e-15)

def summary(values):return {'mean':float(np.mean(values)),'p95':float(np.quantile(values,.95)),'p99':float(np.quantile(values,.99)),'max':float(np.max(values))}

 check('exact cold desktop compressed payload agrees with quality epoch '+ident,z['payload_sha256']==e['payload_sha256'] and z['payload_bytes']==e['payload_bytes']==mip_bytes(z['width'],z['height'],16 if z['normal'] else 8) and a['payload_bytes']==mip_bytes(a['width'],a['height'],16))
 counts[str(fmt)]=counts.get(str(fmt),0)+1
 source=R/ident.removeprefix('res://');check('actual source image is native before-bound '+ident,H(source.read_bytes())==load(B/'cold-exact-probe/receipt.json')['inputs_before'][ident.removeprefix('res://')])
 if z['active']:active_old+=a['payload_bytes'];active_new+=z['payload_bytes']
 q={'resource':ident,'active':z['active'],'normal':z['normal'],'levels':{}}
 for level in [0,3]:
  s=decode('baseline-rgba',a[f'source_mip{level}']);v=decode('baseline-rgba',a[f'decoded_mip{level}']);w=decode('candidate-migration',z[f'decoded_mip{level}'])
  check('opaque source/native decode and equal size '+ident+f' mip{level}',s.shape==v.shape==w.shape and bool(np.all(s[...,3]==255) and np.all(v[...,3]==255) and np.all(w[...,3]==255)))
  entry={'baseline_source_RGB8_absolute':summary(np.abs(v[...,:3].astype(np.int16)-s[...,:3].astype(np.int16))),'candidate_source_RGB8_absolute':summary(np.abs(w[...,:3].astype(np.int16)-s[...,:3].astype(np.int16))),'candidate_baseline_RGB8_absolute':summary(np.abs(w[...,:3].astype(np.int16)-v[...,:3].astype(np.int16)))}
  if z['normal']:
   ns=normal(s);nv=normal(v);nw=normal(w)
   entry['positive_Z_normal_angle_degrees']={name:summary(np.degrees(np.arctan2(np.linalg.norm(np.cross(x,y),axis=-1),np.sum(x*y,axis=-1)))) for name,x,y in [('baseline_source',nv,ns),('candidate_source',nw,ns),('candidate_baseline',nw,nv)]}
  q['levels'][str(level)]=entry
 quality.append(q)

 'quality':quality,'visual_review':{'personally_opened':['pavement','asphalt','sandstone','grass','wood'],'finding':'Source detail and orientation remain recognizable.  'proposal_manifest_sha256':H((B/'proposal-manifest.json').read_bytes()),'exact_native_receipt_sha256':H((B/'cold-exact-probe/receipt.json').read_bytes())}
(O/'independent-review.json').write_text(json.dumps(report,indent=2)+'\n');shutil.copyfile(__file__,O/'independent-audit.py.txt')
