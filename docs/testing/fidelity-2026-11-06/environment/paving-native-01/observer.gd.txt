extends "res://tools/environment_water_visual_controls.gd"
## Matched visual A/B/A only; fresh allocation control owns the15-map budget.
const ContextSky = preload("res://game/levels/horizon_background.gd")
const Library = preload("res://game/levels/art/material_library.gd")
var _paving_meshes: Array[MeshInstance3D] = []
var _paving_contracts: Dictionary = {}

func _run() -> void:
	var args := OS.get_cmdline_user_args()
	if args.size()!=1 or DisplayServer.get_name()=="headless":quit(2);return
	_output=args[0]
	if not (ProjectSettings.globalize_path(_output).simplify_path()+"/").begins_with(ProjectSettings.globalize_path("res://.local/").simplify_path()+"/"):
		quit(2);return
	_started=Time.get_ticks_usec()
	Engine.max_fps=0
	root.size=Vector2i(640,400)
	root.content_scale_size=root.size
	var lighting:=World.new()
	lighting._build_lighting()
	for child: Node in lighting.get_children():
		lighting.remove_child(child)
		root.add_child(child)
		if child is WorldEnvironment:
			var panorama:=ShaderMaterial.new()
			panorama.shader=ContextSky.SKY_SHADER
			panorama.set_shader_parameter("context_sky",ContextSky.SKY_CUBE)
			child.environment.sky.sky_material=panorama
			child.environment.sky.process_mode=Sky.PROCESS_MODE_QUALITY
	lighting.free()
	var materials: Dictionary={}
	for key: String in ["stone","stone_light","sandstone","paving","path","asphalt","wood","glass","window_dull","window_lit","shop_glass","iron"]:
		materials[key]=StandardMaterial3D.new()
	Library.apply(materials)
	var candidate: StandardMaterial3D=materials.paving
	var original: Dictionary=JSON.parse_string(FileAccess.get_file_as_string("res://tools/fixtures/environment/paving_before_source.json"))
	original.files.albedo="res://game/art/assets/environment_pbr/derived/pavement_albedo.png"
	var baseline:=StandardMaterial3D.new()
	Library._apply(baseline,original,Color("d5d9d9"),0.18)
	var ground:=MeshInstance3D.new()
	var plane:=PlaneMesh.new()
	plane.size=Vector2(64,64)
	ground.mesh=plane
	root.add_child(ground)
	_control_water=ground
	_paving_meshes.append(ground)
	# A declared fixture ledge exposes top/vertical triplanar joins; it has no
	# gameplay collision and is not represented as a completed actual route.
	var ledge:=MeshInstance3D.new()
	var box:=BoxMesh.new()
	box.size=Vector3(4,0.55,2.5)
	ledge.mesh=box
	ledge.position=Vector3(3,0.275,1)
	root.add_child(ledge)
	_paving_meshes.append(ledge)
	_control_materials=[{"id":"fan-baseline-a","material":baseline},
		{"id":"setts-candidate","material":candidate},{"id":"fan-baseline-b","material":baseline}]
	_paving_contracts={"fan-baseline-a":_material_record(baseline,1.5),"setts-candidate":_material_record(candidate,2.0),"fan-baseline-b":_material_record(baseline,1.5)}
	_camera=Camera3D.new()
	_camera.fov=68
	_camera.far=220
	root.add_child(_camera)
	_camera.current=true
	_create_pixel_proofs()
	_write_report(false,"startup")
	for mode_index: int in _control_materials.size():
		var mode: Dictionary=_control_materials[mode_index]
		for mesh: MeshInstance3D in _paving_meshes:mesh.material_override=mode.material
		var view_index:=0
		for view: Dictionary in _views():
			_place(view)
			_position_proofs(mode_index,view_index)
			await _record_visual(mode,view)
			var recorded: Dictionary=_images.back()
			recorded.material_contract=_paving_contracts[mode.id]
			_write_report(false,"visual-paving")
			view_index+=1
	_check(_images.size()==9,"all nine matched paving A/B/A originals completed")
	_write_report(true,"complete")
	print("ENVIRONMENT_PAVING_VISUAL_RESULT checks=",_checks," failures=",_failures," images=",_images.size())
	quit(1 if _failures else 0)

func _views() -> Array[Dictionary]:
	return [{"id":"route-eye","at":Vector3(0,1.7,-8),"target":Vector3(0,0,7)},
		{"id":"oblique-close","at":Vector3(-1.2,1.3,-2),"target":Vector3(1.5,0,3)},
		{"id":"wide-repeat","at":Vector3(0,6,-11),"target":Vector3(0,0,10)}]

func _material_record(material: StandardMaterial3D,width: float) -> Dictionary:
	var maps: Array[Dictionary]=[]
	for texture: Texture2D in [material.albedo_texture,material.normal_texture,material.roughness_texture]:
		var image:=texture.get_image()
		_check(image!=null,"native paving texture is inspectable")
		if image:maps.append({"path":texture.resource_path,"width":image.get_width(),"height":image.get_height(),"format":image.get_format(),"compressed":image.is_compressed(),"mipmaps":image.get_mipmap_count(),"native_payload_bytes":image.get_data().size()})
	_check(material.normal_enabled and material.ao_enabled and material.uv1_world_triplanar,"paving retains native normal/AO/world triplanar response")
	_check(material.uv1_scale.is_equal_approx(Vector3.ONE/width),"paving uses documented physical metre scale")
	return {"physical_width_m":width,"maps":maps,"normal_strength":material.normal_scale,"tint":[material.albedo_color.r,material.albedo_color.g,material.albedo_color.b],"texture_filter":material.texture_filter}

func _write_report(complete: bool,phase: String) -> void:
	var report: Dictionary={"schema":1,"complete":complete,"phase":phase,"checks":_checks,"failures":_failures,
		"workload":"paving-material-component-A/B/A","engine":Engine.get_version_info(),"renderer":RenderingServer.get_current_rendering_method(),"adapter":RenderingServer.get_video_adapter_name(),"resolution":[root.size.x,root.size.y],
		"lighting_source":"Exact production GlasgowSquare._build_lighting() plus ordinary-Main Horizon.SKY_SHADER/SKY_CUBE quality setup","geometry":"Declared64m plane and4x0.55x2.5m fixture ledge; no gameplay/collision or actual map route claim",
		"images":_images,"cost_brackets":[],"timing_measured":false,"material_contracts":_paving_contracts,
		"source_provenance":"paving-provenance.json and candidate environment_pbr/manifest.json",
		"limits":"Visual-only640x400 matched cameras and native red/green world markers/Canvas frame proof. Concurrent root untimed headless suite may run; no timing/exclusivity/performance claim. Baseline and candidate maps temporarily coexist for comparison, so visual-process memory is not final production admission. The separate fresh native allocation process must establish the unchanged15bound maps/32MiB contract before promotion. Actual Main route/chase/composition and moving-surface/H1 review remain separate."}
	var file:=FileAccess.open(_output.path_join("visual-paving-report.json"),FileAccess.WRITE)
	if file:file.store_string(JSON.stringify(report,"\t")+"\n");file.close()
	else:_check(false,"paving visual report writable")
