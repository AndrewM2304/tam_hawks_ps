#!/usr/bin/env python3
"""Independent, read-only PNG/camera/source audit of native water controls."""
import hashlib
import json
import math
from pathlib import Path

import numpy as np
from PIL import Image

ROOT = Path('/workspace/tam_hawks_ps')
EVIDENCE = ROOT / '.local/environment-phase2/water-lighting-source/.local/environment-water/shadow-controls-01'
SOURCE = ROOT / '.local/environment-phase2/water-lighting-source'
checks = []


def check(name, condition, actual=None):
    checks.append({'name': name, 'pass': bool(condition), 'actual': actual})


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def read(name):
    return json.loads((EVIDENCE / name).read_text())


receipt = read('receipt.json')
report = read('visual-water-report.json')
before = read('source-before.json')
after = read('source-after.json')
check('source inventories match exactly', before == after, len(before))
source_mismatch = [k for k,v in before.items() if not (SOURCE/k).is_file() or sha(SOURCE/k)!=v]
check('executed private source still matches every recorded input', not source_mismatch, source_mismatch)
check('receipt binds inventory bytes', sha(EVIDENCE/'source-before.json') == sha(EVIDENCE/'source-after.json') == receipt['source_inventory_sha256'])
check('completed clean native capture', receipt['engine_exit']==0 and not receipt['engine_errors'] and receipt['source_stable'] and receipt['eligible'] and report['complete'] and report['checks']==58 and report['failures']==0)
check('raw engine has no errors', all(x not in (EVIDENCE/'engine.log').read_text() for x in ['SCRIPT ERROR:', 'ERROR:', 'Traceback','Parse Error']))
check('no timing claimed', report['timing_measured'] is False and report['cost_brackets']==[] and receipt['timing_measured'] is False)
base=report['lighting_interventions'][0]
for row in report['lighting_interventions']:
    expected={**base,'mode':row['mode']}
    if row['mode']=='native-water-caster-off':expected['water_cast_shadow']=0
    if row['mode']=='native-sun-shadows-off':expected['sun_shadow_enabled']=False
    check('exact one-flag intervention '+row['mode'], row==expected, row)
check('real baseline water and sun both cast shadows', base['water_cast_shadow']==1 and base['sun_shadow_enabled'] is True)
check('actual final ordinary-Main sky identities match source', base['sky_shader_sha256']==before['game/art/assets/horizon/horizon_sky.gdshader'] and base['sky_cube_source_sha256']==before['game/art/assets/horizon/glasgow_horizon_cube.png'])
images = {}
image_results = []
max_projection_error = 0
max_centroid_metadata_error = 0
max_canvas_error = 0
for item in report['images']:
    path = EVIDENCE / item['file']
    array = np.asarray(Image.open(path).convert('RGBA'))
    images[item['file']] = array
    check(item['file'] + ': native dimensions and RGBA hash', array.shape == (400, 640, 4) and hashlib.sha256(array.tobytes()).hexdigest() == item['rgba8_sha256'])
    check(item['file'] + ': PNG receipt hash', sha(path) == receipt['images'][item['file']])
    check(item['file'] + ': no process or physics tick changes at readback', item['physics_before'] == item['physics_after'] and item['process_before'] == item['process_after'])
    check(item['file'] + ': declared perspective camera', item['keep_aspect'] == 1 and item['fov'] == 68 and item['far'] == 1000 and abs(item['near'] - .05) < 1e-7)
    basis = np.array(item['camera_basis'], dtype=float).T
    position = np.array(item['camera_position'], dtype=float)
    check(item['file'] + ': finite orthonormal camera basis and forward', np.isfinite(basis).all() and np.isfinite(position).all() and np.max(np.abs(basis.T @ basis - np.eye(3))) < 1e-6 and np.max(np.abs(np.array(item['camera_forward']) + basis[:, 2])) < 1e-8)
    focal = 200 / math.tan(math.radians(item['fov']) / 2)
    markers = []
    for marker in item['pixel_proofs']['markers']:
        local = np.linalg.solve(basis, np.array(marker['world_position']) - position)
        projected = np.array([320 + focal * local[0] / -local[2], 200 - focal * local[1] / -local[2]])
        projection_error = float(np.linalg.norm(projected - marker['projected_xy']))
        max_projection_error = max(max_projection_error, projection_error)
        px, py = marker['projected_xy']
        x0, x1 = max(0, int(px) - 10), min(640, int(px) + 11)
        y0, y1 = max(0, int(py) - 10), min(400, int(py) + 11)
        window = array[y0:y1, x0:x1, :3].astype(float) / 255
        channel = 0 if marker['color'] == 'red' else 1
        others = [i for i in range(3) if i != channel]
        mask = (window[:, :, channel] > .75) & (window[:, :, channel] - window[:, :, others[0]] > .20) & (window[:, :, channel] - window[:, :, others[1]] > .20)
        yy, xx = np.nonzero(mask)
        centroid = np.array([float(xx.mean() + x0 + .5), float(yy.mean() + y0 + .5)])
        centroid_metadata_error = float(np.linalg.norm(centroid - marker['centroid_xy']))
        native_error = float(np.linalg.norm(centroid - projected))
        max_centroid_metadata_error = max(max_centroid_metadata_error, centroid_metadata_error)
        check(item['file'] + ': independently projected ' + marker['color'] + ' marker matches native pixels', projection_error < .001 and len(xx) == marker['pixel_count'] and centroid_metadata_error < .001 and len(xx) >= 8 and native_error <= 2, {'projection_metadata_error_px': projection_error, 'centroid_metadata_error_px': centroid_metadata_error, 'native_error_px': native_error, 'pixel_count': len(xx)})
        markers.append({'color': marker['color'], 'projected_xy': projected.tolist(), 'centroid_xy': centroid.tolist(), 'native_error_px': native_error})
    proof = item['pixel_proofs']
    cx, cy = proof['canvas_sample_xy']
    canvas = array[cy, cx, :3].astype(float) / 255
    canvas_error = float(np.max(np.abs(canvas - proof['canvas_expected_rgb'])))
    max_canvas_error = max(max_canvas_error, canvas_error)
    check(item['file'] + ': Canvas sample matches independent decoded pixel', canvas_error <= 2 / 255 and np.max(np.abs(canvas - proof['canvas_actual_rgb'])) < 1e-6)
    image_results.append({'file': item['file'], 'markers': markers, 'canvas_error': canvas_error})

modes=['native-baseline','native-water-caster-off','native-sun-shadows-off']
views=['river-oblique','high-coverage','grazing']
check('nine mode/view pairs', len(report['images'])==9 and {(i['mode'],i['view']) for i in report['images']}=={(m,v) for m in modes for v in views})
for item in report['images']:
    expected=next(r for r in report['lighting_interventions'] if r['mode']==item['mode']).copy();expected.pop('mode')
    check('per-image actual lighting matches intervention '+item['file'],item['lighting_contract']==expected)
    counters=item['counters']
    check('actual shadow primitives follow caster/light flag '+item['file'], counters['shadow_primitives']==(2 if item['mode']=='native-baseline' else 0) and counters['shadow_draws']==0)
for view in views:
    rows=[next(i for i in report['images'] if i['view']==view and i['mode']==mode) for mode in modes]
    check('same camera each intervention '+view, all(all(r[k]==rows[0][k] for k in ['camera_basis','camera_position','camera_forward','fov','near','far','keep_aspect']) for r in rows))
    check('same native visible primitive/allocation each intervention '+view, all(all(r['counters'][k]==rows[0]['counters'][k] for k in ['visible_primitives','texture_bytes','buffer_bytes']) for r in rows))
report_out={'schema':1,'reviewer':'independent_qa','method':'Read-only independently decoded original PNG RGBA, recomputed native camera projection and marker centroids, source-byte inventories, exact source interventions and native resource counters. No QA engine execution.','checks':len(checks),'failures':sum(not x['pass'] for x in checks),'source_inputs':len(before),'native_images':9,'native_markers':18,'native_canvas_proofs':9,'max_projection_metadata_error_px':max_projection_error,'max_centroid_metadata_error_px':max_centroid_metadata_error,'max_canvas_error':max_canvas_error,'image_proofs':image_results,'observed':'All nine original frames opened. Fine comb/ring striation visibly disappears in high-coverage/grazing when only water casting is disabled, while bright directional reflection remains. Disabling sun shadows also removes bands, with a larger reflected-light change. Oblique changes are less conspicuous.','verdict':'Capture integrity and bounded static native-StandardMaterial caster isolation pass. This supports an isolated self-shadow explanation for these views, rather than assigning the residual solely to wave NORMAL. Exact underlying renderer cause is not fully diagnosed.','limits':['This is an isolated StandardMaterial plane with exact production plane/daylight/final-Main sky; no world geometry or production wave shader is rendered.','No moving shimmer, production Main caster change, physical gameplay, temporal/cost/M5/AAA acceptance.','Wrapper reports no other Godot at startup; absence of a continuous process observer does not establish whole-run exclusivity. No timing measured.','Zero shadow_draws alone is invalid GLES evidence; native shadow_primitives2->0 and actual flag intervention support this narrower conclusion.'],'checks_detail':checks,'executed_source_hashes':{k:v for k,v in before.items() if k.startswith('tools/environment_water_')},'evidence_hashes':{p.name:sha(p) for p in EVIDENCE.iterdir() if p.is_file()}}
(ROOT/'.local/qa-fidelity/water-shadow-independent-review.json').write_text(json.dumps(report_out,indent=2)+'\n')
print(json.dumps({k:v for k,v in report_out.items() if k not in ['checks_detail','image_proofs','evidence_hashes','executed_source_hashes']},indent=2))
if report_out['failures']:
    print(json.dumps([x for x in checks if not x['pass']],indent=2))
raise SystemExit(1 if report_out['failures'] else 0)
