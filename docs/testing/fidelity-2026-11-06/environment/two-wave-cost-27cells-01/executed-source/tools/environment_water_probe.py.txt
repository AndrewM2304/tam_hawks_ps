#!/usr/bin/env python3
"""Run a bounded, source-frozen native water A/B/A diagnostic locally.

No CI, export, network, save writes or personal Music access. Requires an idle
Godot/render slot, Xorg and an already imported project. Results remain .local.
"""
from __future__ import annotations
import argparse
import datetime
import hashlib
import json
import os
from pathlib import Path
import re
import signal
import subprocess
import time

ROOT = Path(__file__).resolve().parents[1]

def identity() -> dict[str, str]:
    paths = [ROOT / "project.godot"]
    for directory in ("game", "tests", "tools"):
        paths.extend(p for p in (ROOT / directory).rglob("*") if p.is_file() and "__pycache__" not in p.parts)
    controls = ROOT / ".local/environment-fidelity/normal2k-control"
    if controls.exists(): paths.extend(p for p in controls.rglob("*") if p.is_file())
    return {p.relative_to(ROOT).as_posix(): hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted(set(paths))}

def stop(process: subprocess.Popen | None) -> None:
    if process is not None and process.poll() is None:
        os.killpg(process.pid, signal.SIGTERM)
        try: process.wait(timeout=10)
        except subprocess.TimeoutExpired:
            os.killpg(process.pid, signal.SIGKILL)
            process.wait(timeout=10)

def classify_native_godot(processes: list[dict], launch_pid: int) -> dict:
    """Bound ownership to the launched process group/session AND ancestry.

    Native engine startup can fork helpers retaining the godot process name.
    Matching command text alone never grants ownership. Zombies cannot consume
    rendering work and are reported separately rather than as live competitors.
    """
    table = {row["pid"]: row for row in processes}
    owned, competing, zombies = [], [], []
    for row in processes:
        if row["name"] != "godot": continue
        if row["state"].startswith("Z"):
            zombies.append(row)
            continue
        ancestors, cursor, seen = [], row["pid"], set()
        while cursor in table and cursor not in seen:
            seen.add(cursor)
            ancestors.append(cursor)
            if cursor == launch_pid: break
            cursor = table[cursor]["ppid"]
        evidence = dict(row, ancestors=ancestors)
        belongs = launch_pid in ancestors and row["pgid"] == launch_pid and row["sid"] == launch_pid
        (owned if belongs else competing).append(evidence)
    return {"owned": owned, "competing": competing, "zombies": zombies}


def native_process_snapshot() -> list[dict]:
    observed = subprocess.run(["ps", "-eo", "pid=,ppid=,pgid=,sid=,stat=,comm="], capture_output=True, text=True, check=True)
    rows = []
    for line in observed.stdout.splitlines():
        fields = line.split(maxsplit=5)
        if len(fields) == 6:
            rows.append({"pid": int(fields[0]), "ppid": int(fields[1]), "pgid": int(fields[2]), "sid": int(fields[3]), "state": fields[4], "name": fields[5]})
    return rows

def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("label", help="new output-directory label")
    parser.add_argument("--renderer", choices=["gl_compatibility", "forward_plus"], default="gl_compatibility")
    parser.add_argument("--isolated-water", action="store_true", help="short 640x400 actual-plane shader diagnostic; excludes Main and allocation control")
    parser.add_argument("--visual-controls", action="store_true", help="visual-only exact-plane material controls; no timing")
    parser.add_argument("--concurrent-headless-pid", type=int, help="explicit root-authorised untimed headless suite PID; visual-only")
    parser.add_argument("--actual-main-water",action="store_true",help="nine actual Main caster A/B/A views plus inherited3synchronisation proofs; no timing")
    parser.add_argument("--basin-buffer-allocation",action="store_true",help="fresh native single-mesh before/create/release buffer control; no timing")
    args = parser.parse_args()
    if args.basin_buffer_allocation and (args.actual_main_water or args.visual_controls or args.isolated_water):raise SystemExit("Select one workload")
    if args.actual_main_water and (args.visual_controls or args.isolated_water):raise SystemExit("Select one workload")
    if args.visual_controls and args.isolated_water: raise SystemExit("Choose visual-only or timed isolated mode")
    if args.concurrent_headless_pid and not (args.visual_controls or args.actual_main_water or args.basin_buffer_allocation): raise SystemExit("Concurrent suite is permitted only for visual-only controls")
    if not re.fullmatch(r"[A-Za-z0-9_-]+", args.label): raise SystemExit("Invalid label")
    running = subprocess.run(["ps", "-C", "godot", "-o", "pid=,stat=,args="], capture_output=True, text=True)
    concurrent, startup_zombies = [], []
    for line in running.stdout.splitlines():
        fields = line.split(maxsplit=2)
        if len(fields) != 3: continue
        pid, state, command = fields
        if state.startswith("Z"):
            startup_zombies.append({"pid": int(pid), "state": state})
        else:
            concurrent.append(pid + " " + command)
    unexpected = []
    for line in concurrent:
        pid, command = line.split(maxsplit=1)
        if not ((args.visual_controls or args.actual_main_water or args.basin_buffer_allocation) and int(pid) == args.concurrent_headless_pid and " --headless " in " " + command + " " and " --editor " not in " " + command + " " and " --import " not in " " + command + " "): unexpected.append(line)
    if unexpected: raise SystemExit("Unregistered Godot workload: " + repr(unexpected))
    output = ROOT / ".local/environment-water" / args.label
    output.mkdir(parents=True, exist_ok=False)
    before = identity()
    (output / "source-before.json").write_text(json.dumps(before, indent=2) + "\n")
    started = datetime.datetime.now(datetime.timezone.utc).isoformat()
    xorg = process = None
    code = None
    abort_reason = None
    host_samples = []
    active_stage = "display-startup"
    timeout_seconds = 360 if args.actual_main_water else (180 if args.isolated_water else (120 if args.visual_controls or args.basin_buffer_allocation else 1800))
    resolution = "640x400" if args.isolated_water or args.visual_controls else "1280x800"
    common = ["./tools/run.sh", "--rendering-method", args.renderer, "--audio-driver", "Dummy", "--windowed", "--resolution", resolution]
    scripts = ["tools/environment_water_isolated_probe.gd"] if args.isolated_water else ["tools/environment_texture_probe.gd", "tools/environment_water_probe.gd"]
    if args.visual_controls: scripts = ["tools/environment_water_visual_controls.gd"]
    if args.actual_main_water:scripts=["tools/environment_main_basin_topology_capture.gd"]
    if args.basin_buffer_allocation:scripts=["tools/environment_basin_buffer_probe.gd"]
    commands = [common + ["--script", script, "--", str(output)] for script in scripts]
    try:
        with (output / "display").open("w") as display_file, (output / "xorg-console.log").open("w") as xlog:
            xorg = subprocess.Popen(["/usr/lib/xorg/Xorg", "-displayfd", str(display_file.fileno()), "-config", str(ROOT / "tools/xorg-dummy.conf"), "-logfile", str(output / "xorg.log"), "-noreset", "-nolisten", "tcp"], pass_fds=(display_file.fileno(),), stdout=xlog, stderr=subprocess.STDOUT, start_new_session=True)
            display = ""
            for _ in range(50):
                display = (output / "display").read_text().strip()
                if display: break
                if xorg.poll() is not None: raise RuntimeError("Xorg failed")
                time.sleep(0.1)
            if not display: raise RuntimeError("No Xorg display")
            env = dict(os.environ, DISPLAY=":" + display, LIBGL_ALWAYS_SOFTWARE="1", LP_NUM_THREADS="4")
            with (output / "engine.log").open("w") as log:
                for command in commands:
                    active_stage = command[command.index("--script") + 1]
                    process = subprocess.Popen(command, cwd=ROOT, env=env, stdout=log, stderr=subprocess.STDOUT, start_new_session=True)
                    if args.isolated_water:
                        deadline=time.monotonic()+timeout_seconds
                        while process.poll() is None:
                            ownership=classify_native_godot(native_process_snapshot(),process.pid)
                            engines=ownership["owned"]+ownership["competing"]+ownership["zombies"]
                            competitors=ownership["competing"]
                            busy=subprocess.run(["ps","-eo","pid=,comm=,pcpu="],capture_output=True,text=True)
                            other_busy=[]
                            for line in busy.stdout.splitlines():
                                fields=line.split()
                                if len(fields)==3 and fields[1] in {"python","python3","python3.12","blender","ffmpeg"} and int(fields[0])!=os.getpid() and float(fields[2])>=20.0:
                                    other_busy.append({"pid":int(fields[0]),"name":fields[1],"lifetime_cpu_percent":float(fields[2])})
                            host_samples.append({"utc":datetime.datetime.now(datetime.timezone.utc).isoformat(),"elapsed_monotonic_seconds":time.monotonic()-(deadline-timeout_seconds),"launch_pid":process.pid,"owned_native_pids":[row["pid"] for row in ownership["owned"]],"godot_processes":engines,"competing_godot":competitors,"ignored_zombie_pids":[row["pid"] for row in ownership["zombies"]],"other_busy_process_names_only":other_busy,"host_load_average":os.getloadavg()})
                            if competitors:raise RuntimeError("Competing Godot invalidates exclusive cost epoch: "+repr(competitors))
                            if time.monotonic()>=deadline:raise subprocess.TimeoutExpired(command,timeout_seconds)
                            try:process.wait(timeout=0.25)
                            except subprocess.TimeoutExpired:pass
                        code=process.returncode
                    else:
                        code = process.wait(timeout=timeout_seconds)
                    if code: break
    except Exception as error:
        # A slow software-renderer timeout is a rejected epoch, but its native
        # frames/allocation report and exact source identity remain useful.
        abort_reason = type(error).__name__ + ": " + str(error)
    finally:
        stop(process)
        stop(xorg)
    (output / "host-observer.json").write_text(json.dumps({"schema":1,"sampling_interval_seconds":0.25,"samples":host_samples,"limits":"Godot process polling binds owned native workers to launch session/group plus parent ancestry; command text alone is insufficient. Zombies are recorded separately. Shorter processes can escape sampling. Other busy process names/lifetimeCPU and host load are contextual, not an instantaneous CPU exclusivity proof. No unrelated command arguments recorded."},indent=2)+"\n")
    after = identity()
    (output / "source-after.json").write_text(json.dumps(after, indent=2) + "\n")
    changed = sorted(key for key in before.keys() | after.keys() if before.get(key) != after.get(key))
    engine_log = output / "engine.log"
    log = engine_log.read_text() if engine_log.exists() else ""
    engine_errors = bool(re.search(r"SCRIPT ERROR:|SHADER ERROR:|Parse Error:|ERROR:|FAIL:", log))
    reports = {}
    for name in ["water", "texture", "isolated-water", "visual-water", "actual-main-water", "basin-buffer"]:
        report_path = output / (name + "-report.json")
        try:
            reports[name] = json.loads(report_path.read_text()) if report_path.exists() else {}
        except (OSError, ValueError) as error:
            abort_reason = abort_reason or type(error).__name__ + ": " + str(error)
            reports[name] = {}
    report, texture = reports["water"], reports["texture"]
    workload_pass = report.get("failures") == 0 and report.get("complete") is True and len(report.get("images", [])) == 48 and len(report.get("cost_brackets", [])) == 27 and texture.get("failures") == 0 and texture.get("budget_pass") is True
    if args.isolated_water:
        report = reports["isolated-water"]
        workload_pass = report.get("failures") == 0 and report.get("complete") is True and len(report.get("cost_brackets", [])) == 27 and len(report.get("images", [])) == 6
    if args.visual_controls:
        report = reports["visual-water"]
        workload_pass = report.get("failures") == 0 and report.get("complete") is True and len(report.get("images", [])) == 9 and report.get("timing_measured") is False and report.get("cost_brackets") == []
    if args.actual_main_water:
        report=reports["actual-main-water"]
        workload_pass=report.get("failures")==0 and report.get("complete") is True and len(report.get("images",[]))==9 and len(report.get("synchronization_proofs",[]))==3 and report.get("timing_measured") is False
    if args.basin_buffer_allocation:
        report=reports["basin-buffer"]
        workload_pass=report.get("failures")==0 and report.get("complete") is True and report.get("original_resource_released") is True and report.get("budget_pass") is True and report.get("timing_measured") is False
    receipt = {"schema": 1, "started_utc": started, "finished_utc": datetime.datetime.now(datetime.timezone.utc).isoformat(),
        "commands": commands, "git_head": subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True).strip(),
        "source_stable": not changed, "changed_inputs": changed, "engine_exit": code, "engine_errors": engine_errors,
        "abort_reason": abort_reason, "last_stage": active_stage, "process_timeout_seconds": timeout_seconds,
        "workload": "fresh-basin-native-buffer-allocation" if args.basin_buffer_allocation else "actual-main-closed-basin-ABA-1280x800" if args.actual_main_water else ("visual-water-controls-640x400" if args.visual_controls else ("isolated-water-27cells-finalsky-casteroff-640x400" if args.isolated_water else "allocation-plus-full-Main-water-1280x800")),
        "concurrent_headless_authorised_pid": args.concurrent_headless_pid, "other_godot_at_start": concurrent, "ignored_godot_zombies_at_start":startup_zombies, "timing_measured": not (args.visual_controls or args.actual_main_water or args.basin_buffer_allocation),
        "host_observer_samples":len(host_samples),"observed_competing_godot":any(row["competing_godot"] for row in host_samples),
        "source_inventory_sha256": hashlib.sha256((output / "source-after.json").read_bytes()).hexdigest(),
        "environment": {key: env.get(key) for key in ["DISPLAY", "LIBGL_ALWAYS_SOFTWARE", "LP_NUM_THREADS", "VK_ICD_FILENAMES", "VK_DRIVER_FILES", "LD_LIBRARY_PATH"]} if "env" in locals() else {},
        "images": {p.relative_to(output).as_posix(): hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted(output.rglob("*.png"))},
        "eligible": not changed and code == 0 and abort_reason is None and not engine_errors and workload_pass,
        "limits": "Linux diagnostic. Source stability plus no engine errors establishes capture integrity, not visual benefit, target-Mac performance or final material acceptance. Read water-report.json and inspect before/candidate sequences. An aborted epoch remains rejected; completed subsets require separate explicit review."}
    (output / "receipt.json").write_text(json.dumps(receipt, indent=2) + "\n")
    print("ENVIRONMENT_WATER_RECEIPT " + str(output) + " eligible=" + str(receipt["eligible"]))
    if not receipt["eligible"]: raise SystemExit(1)

if __name__ == "__main__": main()
