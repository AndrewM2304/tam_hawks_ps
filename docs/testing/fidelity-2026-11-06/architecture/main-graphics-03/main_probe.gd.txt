extends "res://tools/full_fidelity_capture.gd"
## Prepared author probe; actual normal Main, never the historical component.
## A fresh root engine lease and frozen baseline/candidate/control epoch are needed.
const Batches = preload("res://game/levels/static_batches.gd")
var _architect_records: Array[Dictionary] = []
var _architect_inventory := {}
var _architect_meshes := {}
var _architect_materials := {}

func _run() -> void:
	var args := OS.get_cmdline_user_args()
	if args.size() != 2 or args[1] not in ["inventory", "visual"]:
		quit(2); return
	var graphical := args[1] == "visual"
	if graphical and DisplayServer.get_name() == "headless":
		quit(2); return
	output = ProjectSettings.globalize_path(args[0])
	var ignored := ProjectSettings.globalize_path("res://.local/architectural-fidelity/")
	if not output.begins_with(ignored) or DirAccess.make_dir_recursive_absolute(output) != OK:
		quit(2); return
	root.size = Vector2i(1280,800)
	root.content_scale_size = root.size
	_raw_spool = false
	if graphical: RenderingServer.set_render_loop_enabled(false)
	game = Main.new()
	game.persistence_enabled = false
	_check(not game.has_meta("pt24_design_preview"), "observer assigns no design preview override before normal startup")
	root.add_child(game)
	_check(not game.music._can_control and not game.music._persist, "architecture probe isolates personal Music")
	game.start_mode(true)
	for tick: int in 40:
		await physics_frame
		await process_frame
	await create_timer(0.0,true).timeout
	_check(game.world._station_railway != null
		and game.world._park_pond != null, "architecture probe uses normal default Main with station and pond")
	# Stop descendant simulation only after the untouched ordinary startup.
	game.process_mode = Node.PROCESS_MODE_DISABLED
	_architect_inventory = {"source_triangle_instances":0, "mesh_draw_nodes":0, "geometry_instances":0,
		"mesh_array_serialized_bytes":0,"multimesh_transform_estimate_bytes":0,
		"native_bodies":0,"native_shapes":0,"recess_shells":0,"occluder_owners":[]}
	_architect_scan(game.world)
	_architect_inventory["unique_meshes"] = _architect_meshes.size()
	_architect_inventory["unique_materials"] = _architect_materials.size()
	var contracts := _architect_contracts(game.world)
	var column_contract := _architect_column_contract()
	var aperture_contract := _architect_aperture_contract()
	var material_contract := _architect_material_contract()
	var box_contract := _architect_box_contract(bool(column_contract.get("candidate_module_present",false)))
	if graphical:
		_inspection = Camera3D.new()
		_inspection.name = "DeclaredArchitecturalInspectionCamera"
		_inspection.far = 220
		game.add_child(_inspection)
		await _architect_still("ordinary-spawn", "Ordinary chase after 40 native settled ticks, no pose/camera assignment.")
		await _synchronization_probe()
		game.hud.visible = false
		for view: Array in [
			["exchange-shopfront",Vector3(-29,2,-2),Vector3(-35,2.2,1),68.0],
			["west-upper-oblique",Vector3(-29,4,-2),Vector3(-35,6,5.75),68.0],
			["west-raised-door-shoulder",Vector3(-30.5,1.85,7.5),Vector3(-34.7,1.9,9.59),68.0],
			["gallery-approach",Vector3(7,3.4,-22),Vector3(0,6,-41),68.0],
			["gallery-column-join",Vector3(9,4.7,-35),Vector3(3,5.7,-40.4),68.0],
			["gallery-side-recess",Vector3(15,7,-43),Vector3(9,7.2,-48),68.0],
			["merchant-return",Vector3(-31,4.4,-39),Vector3(-43,7,-46),68.0]]:
			if view[0] not in ["exchange-shopfront","west-upper-oblique","gallery-column-join"]: continue
			_inspection.global_position = view[1]
			_inspection.look_at(view[2])
			_inspection.fov = view[3]
			_inspection.current = true
			await process_frame
			await create_timer(0.0,true).timeout
			await _architect_still(view[0],"Declared static architecture inspection; entire default Main present, native motion frozen after ordinary startup.")
	var report := {"schema":1,"kind":"author_actual_Main_architecture_diagnostic",
		"engine":Engine.get_version_info(),"renderer":RenderingServer.get_current_rendering_method(),
		"adapter":RenderingServer.get_video_adapter_name(),"resolution":[1280,800],
		"production_main":true,"mode":args[1],"checks":checks,"failures":failures,
		"world_inventory":_architect_inventory,"world_contracts":contracts,
		"column_contract":column_contract,"west_upper_aperture":aperture_contract,
		"material_contract":material_contract,"closed_box_contract":box_contract,
		"frames":_architect_records,"sync_proofs":_sync_proofs,
		"limits":"Author evidence only; entire normal Main constructed. Inventory counts all source geometry, not rendered submissions; serialized array bytes and48-byte transform estimates are not allocations. Native buffer/texture/draw counters are same-process observations in static authored views, not pacing evidence. Physics/terrain/routes/grind/gap records allow exact independent before/after comparison. No physical route, final art, timing or M5 acceptance."}
	var file := FileAccess.open(output.path_join("architecture-main.json"),FileAccess.WRITE)
	_check(file != null,"architecture Main receipt writable")
	if file:
		report["checks"] = checks
		report["failures"] = failures
		file.store_string(JSON.stringify(report,"\t")+"\n")
		file.close()
	print("ARCHITECTURE_MAIN checks=",checks," failures=",failures," source_triangles=",_architect_inventory.source_triangle_instances," nodes=",_architect_inventory.mesh_draw_nodes)
	quit(1 if failures else 0)

func _architect_still(label: String, fixture: String) -> void:
	var image: Image = await _render_current_frame()
	var camera := root.get_camera_3d()
	_architect_records.append({"file":label+".png","fixture":fixture,"pixels":_store_frame(image,label+".png"),
		"camera_position":_v(camera.global_position),"camera_forward":_v(-camera.global_basis.z),"camera_fov":camera.fov,
		"player_position":_v(game.skater.global_position),"physics_frame":Engine.get_physics_frames(),
		"process_frame":Engine.get_process_frames(),"draw_calls":int(Performance.get_monitor(Performance.RENDER_TOTAL_DRAW_CALLS_IN_FRAME)),
		"visible_draws":root.get_render_info(Viewport.RENDER_INFO_TYPE_VISIBLE,Viewport.RENDER_INFO_DRAW_CALLS_IN_FRAME),
		"shadow_draws":root.get_render_info(Viewport.RENDER_INFO_TYPE_SHADOW,Viewport.RENDER_INFO_DRAW_CALLS_IN_FRAME),
		"primitives":int(Performance.get_monitor(Performance.RENDER_TOTAL_PRIMITIVES_IN_FRAME)),
		"buffer_bytes":int(Performance.get_monitor(Performance.RENDER_BUFFER_MEM_USED)),
		"texture_bytes":int(Performance.get_monitor(Performance.RENDER_TEXTURE_MEM_USED))})

	# Preserve exact per-image camera/pixels/counters if the global lease stops.
	var progress: FileAccess = FileAccess.open(output.path_join("architecture-main-progress.json"),FileAccess.WRITE)
	if progress:
		progress.store_string(JSON.stringify({"schema":1,"complete":false,"checks":checks,"failures":failures,
			"frames":_architect_records,"sync_proofs":_sync_proofs,"limits":"Partial exact static author frames only; final source/contract receipt may be absent."},"\t")+"\n")
		progress.close()

func _architect_scan(node: Node) -> void:
	if node is PhysicsBody3D: _architect_inventory.native_bodies += 1
	if node is CollisionShape3D: _architect_inventory.native_shapes += 1
	if node is OccluderInstance3D:
		_architect_inventory.occluder_owners.append(str(game.world.get_path_to(node.get_parent())))
	var mesh: Mesh
	var count := 1
	if node is MeshInstance3D: mesh = node.mesh
	elif node is MultiMeshInstance3D and node.multimesh != null:
		mesh = node.multimesh.mesh
		count = node.multimesh.instance_count
		_architect_inventory.multimesh_transform_estimate_bytes += count*48
	if mesh != null:
		_architect_inventory.mesh_draw_nodes += 1
		_architect_inventory.geometry_instances += count
		if node.material_override != null: _architect_materials[node.material_override.get_instance_id()] = true
		var fresh := not _architect_meshes.has(mesh.get_instance_id())
		_architect_meshes[mesh.get_instance_id()] = true
		for surface: int in mesh.get_surface_count():
			var arrays := mesh.surface_get_arrays(surface)
			if arrays.is_empty(): continue
			var indices: Variant = arrays[Mesh.ARRAY_INDEX]
			var vertices: PackedVector3Array = arrays[Mesh.ARRAY_VERTEX]
			_architect_inventory.source_triangle_instances += count*int(indices.size()/3 if indices != null and not indices.is_empty() else vertices.size()/3)
			if node.material_override == null and mesh.surface_get_material(surface) != null:
				_architect_materials[mesh.surface_get_material(surface).get_instance_id()] = true
			if fresh:
				for channel: Variant in arrays:
					if channel != null: _architect_inventory.mesh_array_serialized_bytes += var_to_bytes(channel).size()
		if bool(node.get_meta("architectural_recess_shell",false)): _architect_inventory.recess_shells += 1
	for child: Node in node.get_children(): _architect_scan(child)

func _architect_contracts(world: Node3D) -> Dictionary:
	var collisions := PackedStringArray()
	for node: CollisionShape3D in world.find_children("*","CollisionShape3D",true,false):
		var body := node.get_parent()
		var properties := {}
		for property: Dictionary in node.shape.get_property_list():
			var key := String(property.name)
			if int(property.usage)&PROPERTY_USAGE_STORAGE and key not in ["resource_name","resource_path","resource_local_to_scene","script"]:
				properties[key] = node.shape.get(key)
		collisions.append(var_to_str([str(world.get_path_to(body)),node.global_transform,node.disabled,body.get_class(),body.collision_layer,body.collision_mask,node.shape.get_class(),properties]))
	collisions.sort()
	var rails := PackedStringArray()
	for line: RefCounted in world.get_grind_lines(): rails.append(var_to_str([line.id,line.kind,line.points,line.is_valid]))
	rails.sort()
	return {"collision_records":collisions,"grind_records":rails,
		"route_records":var_to_str(world.get_route_definitions()),"gap_records":var_to_str(world.get_gap_definitions()),
		"terrain_patches":var_to_str(world._terrain.patches),"terrain_polygons":var_to_str(world._terrain.polygons)}

func _architect_column_contract() -> Dictionary:
	var path := "res://game/levels/art/gallery_column.gd"
	if not FileAccess.file_exists(path): return {"candidate_module_present":false}
	var library = load(path)
	var mesh: Mesh = library.shared_mesh()
	var arrays := mesh.surface_get_arrays(0)
	var vertices: PackedVector3Array = arrays[Mesh.ARRAY_VERTEX]
	var tangents: PackedFloat32Array = arrays[Mesh.ARRAY_TANGENT]
	var handedness_ok := vertices.size() == 502 and tangents.size() == 2008
	for index: int in vertices.size():
		var expected := 1.0 if index >= 450 and index < 476 else -1.0
		if tangents.size() < index*4+4 or absf(tangents[index*4+3]-expected)>0.00001:
			handedness_ok = false
	var batches := 0
	var instances := 0
	var cpu_records := 0
	for node: MultiMeshInstance3D in game.world.find_children("*","MultiMeshInstance3D",true,false):
		if node.multimesh != null and node.multimesh.mesh == mesh:
			batches += 1
			instances += node.multimesh.instance_count
			cpu_records += Batches.authored_transforms(node).size()
	_check(handedness_ok,"actual candidate column tangent upload preserves the corrected UV basis")
	_check(instances == 8 and cpu_records == 8,"all eight actual profiled columns preserve authored CPU batch records")
	return {"candidate_module_present":true,"vertices":vertices.size(),"correct_uv_handedness":handedness_ok,
		"runtime_batches":batches,"runtime_instances":instances,"cpu_authored_records":cpu_records,
		"native_aabb":var_to_str(mesh.get_aabb())}

func _architect_aperture_contract() -> Dictionary:
	for building: Node in game.world.get_children():
		if not building is Node3D or not building.position.is_equal_approx(Vector3(-41,0,0)):
			continue
		var wall := building.get_node_or_null("Facade") as Node3D
		if wall == null: continue
		for node: Node in wall.get_children():
			if not node is MeshInstance3D: continue
			var a: Vector3 = node.global_transform.affine_inverse()*(building.global_transform*Vector3(0,5.25,6.5))
			var b: Vector3 = node.global_transform.affine_inverse()*(building.global_transform*Vector3(0,5.25,5.5))
			var faces: PackedVector3Array = node.mesh.get_faces()
			var hits := 0
			for index: int in range(0,faces.size(),3):
				if Geometry3D.segment_intersects_triangle(a,b,faces[index],faces[index+1],faces[index+2]) != null:
					hits += 1
			var cut := bool(node.get_meta("architectural_recess_shell",false))
			_check(hits == (0 if cut else 1),"actual west wall triangle witness agrees with its cut-shell identity")
			return {"building":str(game.world.get_path_to(building)),"mesh_class":node.mesh.get_class(),
				"cut_shell":cut,"triangle_hits_through_upper_opening":hits}
	_check(false,"actual west target wall exists for native aperture witness")
	return {}


func _architect_material_contract() -> Dictionary:
	var materials: Dictionary = game.world._materials
	var keys_by_id := {}
	for key: String in materials: keys_by_id[materials[key].get_instance_id()] = key
	var static_keys: Array[String] = ["facade_glass","window_dull","window_lit"]
	for key: String in static_keys:
		var pane: Material = materials[key]
		_check(pane is ShaderMaterial and pane.shader == Batches.STATIC_WINDOW_SHADER and pane.next_pass == null,
			"native static pane retains exact audited shader identity: "+key)
	_check(materials.glass is StandardMaterial3D,"moving train glass retains its native material class")
	var bindings: Array[Dictionary] = []
	var maps := {}
	var train_glass_bindings := 0
	var room_train_bindings := 0
	var seen := {}
	for node: GeometryInstance3D in game.world.find_children("*","GeometryInstance3D",true,false):
		var mesh: Mesh
		if node is MeshInstance3D: mesh = node.mesh
		elif node is MultiMeshInstance3D and node.multimesh != null: mesh = node.multimesh.mesh
		if mesh == null: continue
		var effective: Array[Material] = []
		if node.material_override != null: effective.append(node.material_override)
		else:
			for surface: int in mesh.get_surface_count():
				var override: Material = node.get_surface_override_material(surface) if node is MeshInstance3D else null
				effective.append(override if override != null else mesh.surface_get_material(surface))
		for material: Material in effective:
			if material == null: continue
			var key := str(keys_by_id.get(material.get_instance_id(),material.resource_name))
			var path := str(game.world.get_path_to(node))
			var room: bool = material is ShaderMaterial and material.shader == Batches.STATIC_WINDOW_SHADER
			if key in static_keys or "ScenicTwoCoachTrain" in path:
				bindings.append({"node":path,"key":key,"class":material.get_class(),"exact_static_shader":room,
					"native_legacy_glass":material == materials.glass})
			if "ScenicTwoCoachTrain" in path:
				if material == materials.glass: train_glass_bindings += 1
				if room: room_train_bindings += 1
			if seen.has(material.get_instance_id()): continue
			seen[material.get_instance_id()] = true
			for property: Dictionary in material.get_property_list():
				if (int(property.usage)&PROPERTY_USAGE_STORAGE) == 0: continue
				var value: Variant = material.get(property.name)
				if value is Texture2D:
					var map_key := str(value.get_instance_id())
					if not maps.has(map_key):
						maps[map_key] = {"first_material_key":key,"first_property":str(property.name),"path":value.resource_path,
							"class":value.get_class(),"width":value.get_width(),"height":value.get_height()}
	_check(train_glass_bindings > 0 and room_train_bindings == 0,"actual moving train binds native glass and no static room shader")
	return {"bindings":bindings,"unique_used_materials":seen.size(),"unique_bound_maps":maps.size(),"bound_maps":maps.values(),
		"moving_train_glass_bindings":train_glass_bindings,"moving_train_static_room_bindings":room_train_bindings,
		"scope":"Native effective material/resource observations, no texture readback or allocation estimate. Native GPU counters are captured separately after real frames."}


func _architect_box_contract(candidate: bool) -> Array[Dictionary]:
	var records: Array[Dictionary] = []
	var expected: Array[Vector3] = [Vector3(-24,0,48),Vector3(0,0,48),Vector3(24,0,48),Vector3(0,6,-134),Vector3(56,0.65,7.5)]
	for occluder: OccluderInstance3D in game.world.find_children("*","OccluderInstance3D",true,false):
		var body := occluder.get_parent() as Node3D
		var building := body.get_parent() as Node3D
		var boxes := 0
		for node: Node in body.get_children():
			if not node is MeshInstance3D or not node.mesh is BoxMesh or not node.is_visible_in_tree(): continue
			if node.cast_shadow == GeometryInstance3D.SHADOW_CASTING_SETTING_SHADOWS_ONLY: continue
			var material := node.material_override as BaseMaterial3D
			if material == null or material.transparency != BaseMaterial3D.TRANSPARENCY_DISABLED: continue
			var shared_key := ""
			for key: String in game.world._materials:
				if game.world._materials[key] == material: shared_key = key
			var correct: bool = occluder.occluder is BoxOccluder3D and not shared_key.is_empty() \
				and occluder.transform.is_equal_approx(node.transform) \
				and occluder.occluder.size.is_equal_approx(node.mesh.size-Vector3.ONE*0.2)
			_check(correct,"actual closed-wall occluder matches one shared opaque visible BoxMesh and its inset")
			boxes += 1
			records.append({"owner":str(game.world.get_path_to(body)),"building_origin":_v(building.position),
				"mesh_class":node.mesh.get_class(),"mesh_size":_v(node.mesh.size),"material_key":shared_key,
				"inset_size":_v(occluder.occluder.size),"exact_transform_and_inset":correct})
		_check(boxes == 1,"each actual occluder owner has exactly one eligible visible Box")
		if candidate:
			var matched := -1
			for index: int in expected.size():
				if building.position.is_equal_approx(expected[index]): matched = index
			_check(matched >= 0,"candidate retains only exact arcade/civic/repair-hall owner positions")
			if matched >= 0: expected.remove_at(matched)
	_check(records.size() == (5 if candidate else 21) and (not candidate or expected.is_empty()),
		"actual closed Box owner count and complete candidate identities agree")
	return records
