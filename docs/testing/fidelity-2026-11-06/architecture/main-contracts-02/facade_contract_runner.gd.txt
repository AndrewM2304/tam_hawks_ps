extends SceneTree
const Facade = preload("res://tests/facade_bay_tests.gd")
var checks := 0
var failures := 0
func _initialize() -> void: _run.call_deferred()
func _check(ok: bool, label: String) -> void:
	checks += 1
	if not ok:
		failures += 1
		push_error("FAIL: "+label)
func _run() -> void:
	await Facade.run(self,_check)
	var out := FileAccess.open("res://.local/architectural-fidelity/facade-contract-result.json",FileAccess.WRITE)
	if out: out.store_string(JSON.stringify({"checks":checks,"failures":failures,"scope":"Private precise FacadeBay contract cohort; not a full regression or visual/cost pass"},"\t")+"\n");out.close()
	print("FACADE_CONTRACT_RESULT checks=",checks," failures=",failures)
	quit(1 if failures else 0)
