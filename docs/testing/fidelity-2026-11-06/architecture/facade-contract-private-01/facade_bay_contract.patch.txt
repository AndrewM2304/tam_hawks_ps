+static func _closed_owner_identity(world: Node3D) -> bool:
+	var expected: Array = [
+		[Vector3(-24,0,48),Vector3(18,15,13.2),"stone_light"],
+		[Vector3(0,0,48),Vector3(18,15,13.2),"stone_light"],
+		[Vector3(24,0,48),Vector3(18,15,13.2),"stone_light"],
+		[Vector3(0,6,-134),Vector3(62,28,12),"stone_light"],
+		[Vector3(56,0.65,7.5),Vector3(15,4.2,12),"warehouse_brick"]]
+	for node: OccluderInstance3D in world.find_children("*","OccluderInstance3D",true,false):
+		var body := node.get_parent() as Node3D
+		if body == null or not body.get_parent() is Node3D: return false
+		var building := body.get_parent() as Node3D
+		var index := -1
+		for candidate: int in expected.size():
+			if building.position.is_equal_approx(expected[candidate][0]): index = candidate
+		if index < 0 or not node.occluder is BoxOccluder3D: return false
+		var contract: Array = expected[index]
+		var matches := 0
+		for child: Node in body.get_children():
+			if not child is MeshInstance3D or not child.mesh is BoxMesh or not child.is_visible_in_tree(): continue
+			if child.cast_shadow == GeometryInstance3D.SHADOW_CASTING_SETTING_SHADOWS_ONLY: continue
+			var material := child.material_override as BaseMaterial3D
+			if material == null or material != world._materials[contract[2]] or material.transparency != BaseMaterial3D.TRANSPARENCY_DISABLED: continue
+			if child.mesh.size.is_equal_approx(contract[1]) and node.transform.is_equal_approx(child.transform) \
+				and node.occluder.size.is_equal_approx(child.mesh.size-Vector3.ONE*0.2): matches += 1
+		if matches != 1: return false
+		expected.remove_at(index)
+	return expected.is_empty()
