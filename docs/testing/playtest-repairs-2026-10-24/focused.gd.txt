extends SceneTree
const Main=preload("res://game/main.gd")
const CameraTests=preload("res://tests/camera_turn_review_tests.gd")
const Classic=preload("res://tests/classic_tricks_integration_tests.gd")
const Named=preload("res://tests/named_grind_integration_tests.gd")
const Repertoire=preload("res://tests/full_repertoire_tests.gd")
const Authentic=preload("res://tests/authentic_fidelity_integration_tests.gd")
const Repairs=preload("res://tests/owner_playtest_repairs_tests.gd")
const Lines=preload("res://tests/compact_clyde_lines_tests.gd")
var passed:=0
var failed:=0
func _initialize() -> void:call_deferred("run")
func check(ok: bool,label: String) -> void:
	if ok:passed+=1;print("PASS: ",label)
	else:failed+=1;print("FAIL: ",label)
func run() -> void:
	var cases:=OS.get_cmdline_user_args()
	if cases.is_empty() or cases.has("camera"):await CameraTests.run(self,check)
	if cases.is_empty() or cases.has("flip"):await Classic._repeat_flip(self,check)
	if cases.is_empty() or cases.has("named"):await Named.run(self,check)
	var game:=Main.new();game.persistence_enabled=false;root.add_child(game);game.start_mode(true)
	if cases.is_empty() or cases.has("repertoire"):await Repertoire.run_playable(self,game,check)
	if cases.is_empty() or cases.has("park"):await Authentic.run(self,game,check,Callable(),["facility_skating"])
	if cases.has("lines"):await Lines.run(self,game,check)
	if cases.has("repairs"):await Repairs.run(self,game,check)
	game.free()
	print("RESULT: %d passed, %d failed"%[passed,failed]);quit(int(failed>0))
