extends "res://tools/rider_asset_inspect.gd"
## Private fixture only; expects the one-scalar Anatomy candidate installed in
## its OWN project copy. The old neutral control is an explicit instance value.
var _environment:WorldEnvironment
var _key:DirectionalLight3D
var _fill:DirectionalLight3D
func _run()->void:
	if DisplayServer.get_name()=="headless":quit(2);return
	root.size=Vector2i(1280,800)
	source_before=source_manifest()
	DirAccess.make_dir_recursive_absolute(OUTPUT+"/synchronization")
	Classic.Setup.register_actions()
	fixture=await Classic._fixture(self)
	var scene:Node3D=fixture.root
	_environment=WorldEnvironment.new()
	_environment.environment=Environment.new()
	scene.add_child(_environment)
	_key=DirectionalLight3D.new()
	_key.shadow_enabled=true
	scene.add_child(_key)
	_fill=DirectionalLight3D.new()
	scene.add_child(_fill)
	var floor_node:=MeshInstance3D.new()
	var plane:=PlaneMesh.new()
	plane.size=Vector2(200,200)
	floor_node.mesh=plane
	var floor_material:=StandardMaterial3D.new()
	floor_material.albedo_color=Color("38434b")
	floor_material.roughness=.92
	floor_node.material_override=floor_material
	floor_node.position.y=-.002
	scene.add_child(floor_node)
	camera=Camera3D.new()
	camera.physics_interpolation_mode=Node.PHYSICS_INTERPOLATION_MODE_OFF
	scene.add_child(camera)
	camera.current=true
	for color:Color in [Color.RED,Color.GREEN,Color.BLUE]:
		var marker:=MeshInstance3D.new()
		var sphere:=SphereMesh.new()
		sphere.radius=1.0
		sphere.height=2.0
		marker.mesh=sphere
		var material:=StandardMaterial3D.new()
		material.shading_mode=BaseMaterial3D.SHADING_MODE_UNSHADED
		material.albedo_color=color
		material.no_depth_test=true
		marker.material_override=material
		marker.physics_interpolation_mode=Node.PHYSICS_INTERPOLATION_MODE_OFF
		marker.visible=false
		scene.add_child(marker)
		sync_markers.append(marker)
	await Classic._frames(self,30)
	# Actor mode, rather than only parent mode, holds all inherited pose branches.
	fixture.rider.process_mode=Node.PROCESS_MODE_DISABLED
	var rig:RefCounted=fixture.rider._visual_rig
	var hand:Node3D=rig._arms[0].hand
	var mesh:MeshInstance3D=hand.get_node("HandAndFingers")
	var cache_mesh_id:=mesh.mesh.get_instance_id()
	var source_arrays:=var_to_bytes([mesh.mesh.surface_get_arrays(0),(mesh.mesh as ArrayMesh).surface_get_blend_shape_arrays(0)])
	for lighting:String in ["production","studio"]:
		_set_lighting(lighting)
		for variant:String in ["control022","candidate038"]:
			for arm:Dictionary in rig._arms:
				rig._anatomy.pose_hand(arm.hand,0.0)
				if variant=="control022":(arm.hand.get_node("HandAndFingers")as MeshInstance3D).set_blend_shape_value(0,.22)
			var center:Vector3=hand.to_global(Vector3(0,-.11,-.01))
			for angle:String in ["palm","side"]:
				var local_at:=Vector3(.11,-.10,-.39)if angle=="palm"else Vector3(.32,-.10,-.13)
				await shot(lighting+"-"+variant+"-"+angle,hand.to_global(local_at),center,35)
				frames[-1]["lighting"]=_light_record(lighting)
				frames[-1]["variant"]=variant
				frames[-1]["hand_mesh_resource_id"]=mesh.mesh.get_instance_id()
				frames[-1]["hand_blend"]=mesh.get_blend_shape_value(0)
				if mesh.mesh.get_instance_id()!=cache_mesh_id or source_arrays!=var_to_bytes([mesh.mesh.surface_get_arrays(0),(mesh.mesh as ArrayMesh).surface_get_blend_shape_arrays(0)]):
					failures+=1
					push_error("Hand scalar fixture changed native source/cache identity")
		var head:Node3D=rig._head
		var face:Vector3=head.to_global(Vector3(0,.02,-.025))
		await shot(lighting+"-nape-control",face+head.global_basis.x*.78,face,35)
		frames[-1]["lighting"]=_light_record(lighting)
	var after:=source_manifest()
	var out:=FileAccess.open(OUTPUT+"/neutral-hand-receipt.json",FileAccess.WRITE)
	out.store_string(JSON.stringify({"scope":"Private hand neutral scalar and matched lighting controls. Source candidate uses actual pose_hand API; baseline deliberately overrides only instance scalar022. All original mesh/target streams and material/maps unchanged. No actualgame source, fullarm/grab integration, normalmotion orAAA pass.","frames":frames,"failures":failures,"source_before":source_before,"source_after":after,"source_stable":source_before==after,"lighting_scope":"Exact declared production parameters versus existing inspector setup in a blanknative-floor fixture; wholecity occlusion/targetdevice remain separate. No skin/nape material adjustment."},"\t"))
	out.close()
	print("NEUTRAL_HAND_STUDIO frames=",frames.size()," failures=",failures)
	quit(1 if failures else 0)
func _set_lighting(which:String)->void:
	var env:Environment=_environment.environment
	env.tonemap_mode=Environment.TONE_MAPPER_ACES
	env.ambient_light_source=Environment.AMBIENT_SOURCE_COLOR
	if which=="production":
		var sky_material:=ProceduralSkyMaterial.new()
		sky_material.sky_top_color=Color("8e9eaa")
		sky_material.sky_horizon_color=Color("c4c8c4")
		sky_material.ground_bottom_color=Color("727b7c")
		sky_material.ground_horizon_color=Color("c4c8c4")
		sky_material.sun_angle_max=5.0
		var sky:=Sky.new()
		sky.sky_material=sky_material
		env.sky=sky
		env.background_mode=Environment.BG_SKY
		env.ambient_light_color=Color("d0dce2")
		env.ambient_light_energy=.36
		env.fog_enabled=true
		env.fog_density=.0012
		env.fog_light_color=Color("b4bec3")
		env.fog_sky_affect=.15
		_key.rotation_degrees=Vector3(-42,-32,0)
		_key.light_color=Color("fffaf3")
		_key.light_energy=.86
		_key.shadow_opacity=.8
		_key.shadow_bias=.1
		_key.shadow_normal_bias=2.0
		_key.directional_shadow_mode=DirectionalLight3D.SHADOW_ORTHOGONAL
		_key.directional_shadow_max_distance=75
		_fill.light_energy=0.0
	else:
		env.sky=null
		env.background_mode=Environment.BG_COLOR
		env.background_color=Color("29333d")
		env.ambient_light_color=Color("b9c9dc")
		env.ambient_light_energy=.62
		env.fog_enabled=false
		_key.rotation_degrees=Vector3(-40,-35,0)
		_key.light_color=Color("fff1dd")
		_key.light_energy=1.6
		_key.shadow_opacity=1.0
		_fill.rotation_degrees=Vector3(-25,145,0)
		_fill.light_color=Color("abc9e9")
		_fill.light_energy=.65
func _light_record(which:String)->Dictionary:
	return {"label":which,"ambient_energy":_environment.environment.ambient_light_energy,"ambient_color":_environment.environment.ambient_light_color,"key_energy":_key.light_energy,"key_color":_key.light_color,"key_rotation_degrees":_key.rotation_degrees,"fill_energy":_fill.light_energy,"fill_color":_fill.light_color,"tonemap":_environment.environment.tonemap_mode,"background":_environment.environment.background_mode,"fog":_environment.environment.fog_enabled}
