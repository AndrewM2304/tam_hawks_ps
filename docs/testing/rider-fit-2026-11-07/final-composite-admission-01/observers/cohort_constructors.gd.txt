extends SceneTree
const Main=preload("res://game/main.gd")
var passes:=0
var failures:Array[String]=[]
var modules:Array=[]
func _initialize()->void:_run.call_deferred()
func _check(ok:bool,label:String)->void:
 if ok:passes+=1
 else:failures.append(label)
 print("PASS " if ok else "FAIL ",label)
func _run()->void:
 var before:=passes
 await load("res://tests/fitted_garment_tests.gd").run(self,_check)
 modules.append({"name":"FittedGarmentNative","passes":passes-before})
 before=passes
 await load("res://tests/activity_actor_tests.gd").run(self,_check)
 modules.append({"name":"ActivityActorFull","passes":passes-before})
 var game:=Main.new()
 _check(not game.persistence_enabled,"final constructor Main isolates persistence before tree")
 game.persistence_enabled=false
 root.add_child(game)
 for tick:int in 40:await physics_frame;await process_frame
 before=passes
 await load("res://tests/activity_actor_tests.gd").run_world(self,game,_check)
 await load("res://tests/ambient_adult_tests.gd").run(self,game,_check)
 await load("res://tests/cat_encounter_tests.gd").run(self,game,_check)
 await load("res://tests/market_encounter_tests.gd").run(self,game,_check)
 modules.append({"name":"DefaultMainNPCConstructorsAndEncounters","passes":passes-before})
 game.queue_free()
 await process_frame
 var out:=FileAccess.open("res://.local/final-constructor-result.json",FileAccess.WRITE)
 out.store_string(JSON.stringify({"passes":passes,"failures":failures,"modules":modules},"\t"));out.close()
 print("FINAL_CONSTRUCTORS ",passes," failures=",failures)
 quit(0 if failures.is_empty() else 1)
