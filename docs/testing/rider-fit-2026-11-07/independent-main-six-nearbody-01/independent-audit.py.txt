    ck(part+' all declared seam source identities authenticated',set(mapped)==ids)
    boundaries[part]=(mapped,positions,bones,weights)

def skinned_points(frame,part,clock):
    mapped,positions,bones,weights=boundaries[part]
    palette=np.asarray(frame['native_rendered']['native_skin_palettes'][part]['palette'])
    model=np.asarray(frame['native_rendered']['native_skin_palettes'][part]['mesh_'+clock])
    out={}
    for sid,indices in mapped.items():
        out[sid]=[]
        for ni in indices:
            # GPU semantics: weighted LOCAL xyz, discard local.w, then MODEL once.
            local=np.zeros(3)
            for bi,w in zip(bones[ni],weights[ni]):local+=(palette[bi]@np.r_[positions[ni],1])[:3]*w
            out[sid].append(model_point(model,local))
    return out
