func _npc_compatibility() -> void:
	board.visible = false
	rider.visible = false
	for index in NPC_SCRIPTS.size():
		var label: String = NPC_LABELS[index]
		var selected := OS.get_cmdline_user_args()
		if not selected.is_empty() and not selected.has(label):continue
		var actor: Node3D = NPC_SCRIPTS[index].new()
		if index < 2:actor.set("steward",index == 1)
		stage.add_child(actor)
		if actor.has_method("set_audio_enabled"):actor.set_audio_enabled(false)
		for tick in 20:
			await physics_frame
			await process_frame
		actor.process_mode = Node.PROCESS_MODE_DISABLED
		active_npc = actor
		var found: MeshInstance3D = actor.find_child("AnatomicalHead",true,false)
		npc_head = found.get_parent()
		npc_skeleton = actor.find_child("ClothingSkeleton",true,false)
		for tick in 3:
			await physics_frame
			await process_frame
		var neck_world := npc_skeleton.global_transform * npc_skeleton.get_bone_global_pose(npc_skeleton.find_bone("neck"))
		var basis_error := maxf(neck_world.basis.x.distance_to(npc_head.global_basis.x),maxf(neck_world.basis.y.distance_to(npc_head.global_basis.y),neck_world.basis.z.distance_to(npc_head.global_basis.z)))
		if neck_world.origin.distance_to(npc_head.global_position) > 0.00005 or basis_error > 0.00005:
			failures += 1
			push_error("Actual NPC Head and native neck bone differ: "+label)
		authored_state = {"actor_script":NPC_SCRIPTS[index].resource_path,"actor_label":label,"native_neck_origin_error_m":neck_world.origin.distance_to(npc_head.global_position),"native_neck_basis_max_axis_error":basis_error,"scope":"Actual actor constructor and settled native pose, then explicitly frozen for source-bound stills. Not Main/input or motion acceptance."}
		var npc_resources := {"triangles":0,"surfaces":0,"mesh_instances":0,"textures":{}}
		_collect_resources(actor,npc_resources)
		for key: Variant in npc_resources.textures:
			shared_texture_resources[key] = npc_resources.textures[key]
		npc_resources.textures = _texture_summary(npc_resources.textures)
		npc_resources.scope = "Actual selected NPC constructor including its role props. Source geometry and texture inventory, not draw calls, allocation or performance. Player65000triangle cap does not apply to an entire actor plus street props."
		npc_resource_inventories[label] = npc_resources
		var target := found.global_position + Vector3(0,-.6,0)
		await shot(label,target+Vector3(1.5,.28,-2.8),target,38)
		actor.queue_free()
		active_npc = null
		npc_head = null
		npc_skeleton = null
		await process_frame

func _pose(state: Dictionary,count: int) -> void:
	authored_state = state.duplicate(true)
	for index in count:
		rig.animate(1.0/60,state)
		await physics_frame
		await process_frame

func _native() -> void:
	RenderingServer.set_render_loop_enabled(true)
	await process_frame
	await RenderingServer.frame_post_draw
	await process_frame
	await RenderingServer.frame_post_draw
	RenderingServer.set_render_loop_enabled(false)

func shot(label: String,from: Vector3,toward: Vector3,fov: float) -> void:
	var selected := OS.get_cmdline_user_args()
	if not selected.is_empty() and not selected.has(label):return
	camera.global_position = from
	camera.look_at(toward)
	camera.fov = fov
	camera.reset_physics_interpolation()
	var depth := from.distance_to(toward)
	var points: Array[Vector3] = [toward,toward+camera.global_basis.x*depth*.07,toward+camera.global_basis.y*depth*.07]
	for index in 3:
		markers[index].global_position = points[index]
		markers[index].scale = Vector3.ONE*depth*.003
		markers[index].visible = true
	var pose_before := _pose_matrices()
	await _native()
	var pose_before_proof_readback := _pose_matrices()
	var proof_tick := {"physics":Engine.get_physics_frames(),"process":Engine.get_process_frames()}
	var proof_image := root.get_texture().get_image()
	var proof_tick_after := {"physics":Engine.get_physics_frames(),"process":Engine.get_process_frames()}
	var pose_after_proof_readback := _pose_matrices()
	proof_image.convert(Image.FORMAT_RGBA8)
	var proof_path := "res://output/synchronization/"+label+".png"
	proof_image.save_png(proof_path)
	for marker in markers:marker.visible = false
	await _native()
	var path := "res://output/"+label+".png"
	var pose_before_clean_readback := _pose_matrices()
	var clean_tick := {"physics":Engine.get_physics_frames(),"process":Engine.get_process_frames()}
	var clean_image := root.get_texture().get_image()
	var allocation := {"texture_bytes":int(Performance.get_monitor(Performance.RENDER_TEXTURE_MEM_USED)),
		"video_bytes":int(Performance.get_monitor(Performance.RENDER_VIDEO_MEM_USED)),
		"buffer_bytes":int(Performance.get_monitor(Performance.RENDER_BUFFER_MEM_USED)),
		"draw_calls":int(Performance.get_monitor(Performance.RENDER_TOTAL_DRAW_CALLS_IN_FRAME)),
		"primitives":int(Performance.get_monitor(Performance.RENDER_TOTAL_PRIMITIVES_IN_FRAME)),
		"scope":"Actual native clean-frame process/driver monitors; includes viewport, stage, shadows and preloaded resources. Not target Mac allocation or performance; no frame-time claim."}
	var clean_tick_after := {"physics":Engine.get_physics_frames(),"process":Engine.get_process_frames()}
	var pose_after_clean_readback := _pose_matrices()
	clean_image.convert(Image.FORMAT_RGBA8)
	clean_image.save_png(path)
	var proofs := []
	for index in 3:
		var projected := camera.unproject_position(points[index]) * Vector2(proof_image.get_size())/root.get_visible_rect().size
		var count := 0
		var total := Vector2.ZERO
		for y in range(int(projected.y)-12,int(projected.y)+13):
			for x in range(int(projected.x)-12,int(projected.x)+13):
				var pixel := proof_image.get_pixel(x,y)
				var clean_pixel := clean_image.get_pixel(x,y)
				var changed := absf(pixel.r-clean_pixel.r)+absf(pixel.g-clean_pixel.g)+absf(pixel.b-clean_pixel.b)>.12
				var channels := [pixel.r,pixel.g,pixel.b]
				if changed and channels[index]>.75 and channels[index]-channels[(index+1)%3]>.2 and channels[index]-channels[(index+2)%3]>.2:
					count += 1
					total += Vector2(x+.5,y+.5)
		var error := (total/count).distance_to(projected) if count else 999.0
		proofs.append({"count":count,"centroid_error":error,"projected":projected,"world":points[index]})
		if count<4 or error>2.5:
			failures += 1
			push_error("Private fit camera pixel synchronization failed "+label)
	if pose_before != _pose_matrices() or pose_before != pose_before_proof_readback or pose_before != pose_after_proof_readback or pose_before != pose_before_clean_readback or pose_before != pose_after_clean_readback or proof_tick != proof_tick_after or clean_tick != clean_tick_after:
		failures += 1
		push_error("Private rig/camera pose changed during native proof/clean draws: "+label)
	frames.append({"label":label,"png_sha256":FileAccess.get_sha256(path),"synchronization":proofs,
		"native_allocation_and_render_counters":allocation,
		"native_rgba_sha256":_image_sha256(clean_image),"synchronization_native_rgba_sha256":_image_sha256(proof_image),"proof_frame_counters":proof_tick,"clean_frame_counters":clean_tick,
		"proof_readback_counter_bracket":[proof_tick,proof_tick_after],"clean_readback_counter_bracket":[clean_tick,clean_tick_after],
		"pose_before_proof_readback":pose_before_proof_readback,"pose_after_proof_readback":pose_after_proof_readback,
		"pose_before_clean_readback":pose_before_clean_readback,"pose_after_clean_readback":pose_after_clean_readback,
		"sync_png_sha256":FileAccess.get_sha256(proof_path),"camera_position":camera.global_position,"fov":camera.fov,"authored_state":authored_state.duplicate(true),"native_state":_pose_matrices()})

func _sources() -> Dictionary:
	var result := {}
	_scan("res://game",result)
	result["res://preview.gd"] = FileAccess.get_sha256("res://preview.gd")
	result["res://project.godot"] = FileAccess.get_sha256("res://project.godot")
	result["res://capture-load.json"] = FileAccess.get_sha256("res://capture-load.json")
	return result

func _scan(path: String,result: Dictionary) -> void:
	for file in DirAccess.get_files_at(path):
		result[path+"/"+file] = FileAccess.get_sha256(path+"/"+file)
	for directory in DirAccess.get_directories_at(path):_scan(path+"/"+directory,result)

func _anatomy_metrics() -> Dictionary:
	var head_mesh: MeshInstance3D = rig._head.get_node("AnatomicalHead")
	var head_min := INF
	var head_max := -INF
	for p: Vector3 in head_mesh.mesh.get_faces():
		var y := (head_mesh.global_transform*p).y
		head_min = minf(head_min,y)
		head_max = maxf(head_max,y)
	var sole_min := INF
	var lengths := []
	var skeleton: Skeleton3D = rig._garments.skeleton
	for leg: Dictionary in rig._legs:
		var sole: MeshInstance3D = leg.foot.get_node("Cupsole")
		for p: Vector3 in sole.mesh.get_faces():
			sole_min = minf(sole_min,(sole.global_transform*p).y)
		var prefix := "left" if leg.side < 0 else "right"
		var hip := skeleton.get_bone_global_pose(skeleton.find_bone(prefix+"_thigh")).origin
		var knee := skeleton.get_bone_global_pose(skeleton.find_bone(prefix+"_calf")).origin
		lengths.append({"side":leg.side,"upper_leg_m":hip.distance_to(knee),"lower_leg_m":knee.distance_to(leg.foot.position)})
	var source: Dictionary = JSON.parse_string(FileAccess.get_file_as_string("res://game/art/assets/makehuman_hm08/manifest.json"))
	return {"scope":"Actual settled private idle is bent-knee skating posture, not standing stature or Main acceptance.",
		"native_head_local_bounds":head_mesh.mesh.get_aabb(),"native_head_world_y_m":[head_min,head_max],
		"actual_lowest_visible_sole_world_y_m":sole_min,"actual_idle_head_to_sole_height_m":head_max-sole_min,
		"actual_leg_lengths":lengths,"authored_arm_lengths_m":[.30,.29],"authored_shoulder_width_m":.43,
		"authored_hip_width_m":.21,"source_revision":source.source_revision,"source_ancestry_adult_identity":source.adaptation.adult_morph,
		"source_head_mesh_sha256":FileAccess.get_sha256("res://game/art/assets/makehuman_hm08/head.obj"),
		"actual_rendered_head_derivative_sha256":FileAccess.get_sha256("res://game/art/rider/fitted_head.obj"),
		"board_source_unchanged_sha256":FileAccess.get_sha256("res://game/art/assets/tam_skateboard/tam_skateboard.glb")}

func _native_inventory() -> Dictionary:
	var result := {"triangles":0,"surfaces":0,"mesh_instances":0,"textures":{}}
	_collect_resources(rider,result)
	_collect_resources(board,result)
	for key: Variant in result.textures:
		shared_texture_resources[key] = result.textures[key]
	var textures := _texture_summary(result.textures)
	result.textures = textures.textures
	result.conservative_rgba_mips_bytes = textures.conservative_rgba_mips_bytes
	result.scope = "Actual instantiated private Rider and native Board. Source triangle count, all active material Texture2D properties; conservative uncompressed RGBA mip estimate, not target GPU allocation or timing."
	if result.triangles>65000 or textures.conservative_rgba_mips_bytes>64*1024*1024:
		failures += 1
		push_error("Private fitted native resource budget exceeded.")
	return result

func _texture_summary(resources: Dictionary) -> Dictionary:
	var details := []
	var conservative_bytes := 0
	for texture: Texture2D in resources.values():
		var bytes := int(texture.get_width()*texture.get_height()*4.0*4.0/3.0)
		conservative_bytes += bytes
		var image := texture.get_image()
		if image == null:
			failures += 1
			push_error("Actual bound texture native readback unavailable: "+texture.resource_path)
		details.append({"path":texture.resource_path,"width":texture.get_width(),"height":texture.get_height(),"conservative_rgba_mips_bytes":bytes,
			"native_format":image.get_format() if image != null else -1,"native_compressed":image.is_compressed() if image != null else false,
			"native_mipmap_count":image.get_mipmap_count() if image != null else -1,"native_readback_payload_bytes":image.get_data().size() if image != null else -1})
	return {"textures":details,"conservative_rgba_mips_bytes":conservative_bytes,"scope":"Unique actual instantiated material textures across selected actors; conservative RGBA with mips. Not measured allocation or timing."}

func _collect_resources(node: Node,result: Dictionary) -> void:
	if not result.has("bindings"):
		result.bindings = []
	if node is MeshInstance3D and node.mesh != null:
		result.mesh_instances += 1
		for surface in node.mesh.get_surface_count():
			result.surfaces += 1
			var arrays: Array = node.mesh.surface_get_arrays(surface)
			var indices: Variant = arrays[Mesh.ARRAY_INDEX]
			result.triangles += indices.size()/3 if indices is PackedInt32Array and not indices.is_empty() else arrays[Mesh.ARRAY_VERTEX].size()/3
			var material: Material = node.get_active_material(surface)
			if material != null:
				for property: Dictionary in material.get_property_list():
					var value: Variant = material.get(property.name)
					if value is Texture2D:
						result.textures[value.get_instance_id()] = value
						result.bindings.append({"mesh_path":str(node.get_path()),"surface":surface,"material_instance_id":material.get_instance_id(),"material_resource_path":material.resource_path,"property":property.name,"texture_path":value.resource_path,"texture_instance_id":value.get_instance_id(),"native_uv1":_material_uv(material)})
	for child in node.get_children():_collect_resources(child,result)

func _transform_rows(value: Transform3D) -> Array:
	return [[value.basis.x.x,value.basis.y.x,value.basis.z.x,value.origin.x],
		[value.basis.x.y,value.basis.y.y,value.basis.z.y,value.origin.y],
		[value.basis.x.z,value.basis.y.z,value.basis.z.z,value.origin.z],[0,0,0,1]]

func _pose_matrices() -> Dictionary:
	var skeleton: Skeleton3D = npc_skeleton if is_instance_valid(npc_skeleton) else rig._garments.skeleton
	var head: Node3D = npc_head if is_instance_valid(npc_head) else rig._head
	var model: Node3D = active_npc if is_instance_valid(active_npc) else rider
	var bones := {}
	for bone in skeleton.get_bone_count():bones[skeleton.get_bone_name(bone)] = _transform_rows(skeleton.get_bone_global_pose(bone))
	var contacts := {}
	_contact_node_poses(model,contacts)
	return {"camera_global":_transform_rows(camera.global_transform),"camera_near":camera.near,"camera_far":camera.far,
		"camera_forward":[-camera.global_basis.z.x,-camera.global_basis.z.y,-camera.global_basis.z.z],"viewport_pixels":[root.size.x,root.size.y],
		"camera_projection_columns":_projection_columns(),"head_global":_transform_rows(head.global_transform),
		"rider_global":_transform_rows(model.global_transform),"skeleton_global":_transform_rows(skeleton.global_transform),"bone_global_pose_in_skeleton":bones,"actual_hand_foot_mesh_transforms_and_blend_shapes":contacts,
		"live_player_jacket_color":_color_channels((rider.get_node("SkinnedJacket") as MeshInstance3D).material_override.albedo_color),
		"debug_mesh_ids":OS.get_cmdline_user_args().has("--mesh-ids"),"id_colours":{"Head":"yellow","UpperChest":"magenta","SkinnedJacket":"cyan"}}

func _contact_node_poses(node: Node,result: Dictionary) -> void:
	if node is MeshInstance3D and node.name in ["HandAndFingers", "Cupsole", "CanvasUpper", "FlatLaces"]:
		var shapes := []
		if node.mesh is ArrayMesh:
			for index in (node.mesh as ArrayMesh).get_blend_shape_count():
				shapes.append(node.get_blend_shape_value(index))
		var materials := []
		for surface in node.mesh.get_surface_count():
			var material: Material = node.get_active_material(surface)
			if material is StandardMaterial3D:
				materials.append({"surface":surface,"native_uv1":_material_uv(material),"albedo_texture":material.albedo_texture.resource_path if material.albedo_texture else "","normal_texture":material.normal_texture.resource_path if material.normal_texture else "","roughness_texture":material.roughness_texture.resource_path if material.roughness_texture else ""})
		result[str(node.get_path())] = {"global_transform":_transform_rows(node.global_transform),"blend_shape_values":shapes,"native_materials":materials}
	for child in node.get_children():
		_contact_node_poses(child,result)

func _image_sha256(value: Image) -> String:
	var context := HashingContext.new()
	context.start(HashingContext.HASH_SHA256)
	context.update(value.get_data())
	return context.finish().hex_encode()

func _projection_columns() -> Array:
	var projection := camera.get_camera_projection()
	var result := []
	for value: Vector4 in [projection.x,projection.y,projection.z,projection.w]:result.append([value.x,value.y,value.z,value.w])
	return result

func _material_uv(material: Material) -> Dictionary:
	var standard := material as StandardMaterial3D
	if standard == null:
		return {"unsupported_material_class":material.get_class()}
	return {"uv1_scale":[standard.uv1_scale.x,standard.uv1_scale.y,standard.uv1_scale.z],"uv1_offset":[standard.uv1_offset.x,standard.uv1_offset.y,standard.uv1_offset.z],"texture_repeat":standard.texture_repeat}
