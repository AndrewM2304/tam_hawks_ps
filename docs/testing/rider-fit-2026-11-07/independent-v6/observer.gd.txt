	world.environment.ambient_light_energy = .62
	world.environment.tonemap_mode = Environment.TONE_MAPPER_ACES

	rig = Rig.new()
	rig.build(board,rider,shirt)

	for index in rig._garments.skeleton.get_bone_count():

func _pose(state: Dictionary,count: int) -> void:
	for index in count:
		rig.animate(1.0/60,state)
		await physics_frame
		await process_frame

func _native() -> void:
	RenderingServer.set_render_loop_enabled(true)
	await process_frame
	await RenderingServer.frame_post_draw
	await process_frame
	await RenderingServer.frame_post_draw
	RenderingServer.set_render_loop_enabled(false)

func shot(label: String,from: Vector3,toward: Vector3,fov: float) -> void:
	var selected := OS.get_cmdline_user_args()
	if not selected.is_empty() and not selected.has(label):return
	camera.global_position = from
	camera.look_at(toward)
	camera.fov = fov
	camera.reset_physics_interpolation()
	var depth := from.distance_to(toward)
	var points: Array[Vector3] = [toward,toward+camera.global_basis.x*depth*.07,toward+camera.global_basis.y*depth*.07]
	for index in 3:
		markers[index].global_position = points[index]
		markers[index].scale = Vector3.ONE*depth*.003
		markers[index].visible = true
	await _native()
	var proof_image := root.get_texture().get_image()
	var proofs := []
	for index in 3:
		var projected := camera.unproject_position(points[index]) * Vector2(proof_image.get_size())/root.get_visible_rect().size
		var count := 0
		var total := Vector2.ZERO
		for y in range(int(projected.y)-12,int(projected.y)+13):
			for x in range(int(projected.x)-12,int(projected.x)+13):
				var pixel := proof_image.get_pixel(x,y)
				var channels := [pixel.r,pixel.g,pixel.b]
				if channels[index]>.75 and channels[index]-channels[(index+1)%3]>.2 and channels[index]-channels[(index+2)%3]>.2:
					count += 1
					total += Vector2(x+.5,y+.5)
		var error := (total/count).distance_to(projected) if count else 999.0
		proofs.append({"count":count,"centroid_error":error,"projected":projected,"world":points[index]})
		if count<4 or error>2.5:
			failures += 1
			push_error("Private fit camera pixel synchronization failed "+label)
	var proof_path := "res://output/synchronization/"+label+".png"
	proof_image.save_png(proof_path)
	for marker in markers:marker.visible = false
	await _native()
	var path := "res://output/"+label+".png"
	root.get_texture().get_image().save_png(path)
	frames.append({"label":label,"png_sha256":FileAccess.get_sha256(path),"synchronization":proofs,
		"sync_png_sha256":FileAccess.get_sha256(proof_path),"camera_position":camera.global_position,"fov":camera.fov})

func _sources() -> Dictionary:
	var result := {}
	_scan("res://game",result)
	result["res://preview.gd"] = FileAccess.get_sha256("res://preview.gd")
	result["res://project.godot"] = FileAccess.get_sha256("res://project.godot")
	return result

func _scan(path: String,result: Dictionary) -> void:
	for file in DirAccess.get_files_at(path):
		result[path+"/"+file] = FileAccess.get_sha256(path+"/"+file)
	for directory in DirAccess.get_directories_at(path):_scan(path+"/"+directory,result)

func _anatomy_metrics() -> Dictionary:
	var head_mesh: MeshInstance3D = rig._head.get_node("AnatomicalHead")
	var head_min := INF
	var head_max := -INF
	for p: Vector3 in head_mesh.mesh.get_faces():
		var y := (head_mesh.global_transform*p).y
		head_min = minf(head_min,y)
		head_max = maxf(head_max,y)
	var sole_min := INF
	var lengths := []
	var skeleton: Skeleton3D = rig._garments.skeleton
	for leg: Dictionary in rig._legs:
		var sole: MeshInstance3D = leg.foot.get_node("Cupsole")
		for p: Vector3 in sole.mesh.get_faces():
			sole_min = minf(sole_min,(sole.global_transform*p).y)
		var prefix := "left" if leg.side < 0 else "right"
		var hip := skeleton.get_bone_global_pose(skeleton.find_bone(prefix+"_thigh")).origin
		var knee := skeleton.get_bone_global_pose(skeleton.find_bone(prefix+"_calf")).origin
		lengths.append({"side":leg.side,"upper_leg_m":hip.distance_to(knee),"lower_leg_m":knee.distance_to(leg.foot.position)})
	var source: Dictionary = JSON.parse_string(FileAccess.get_file_as_string("res://game/art/assets/makehuman_hm08/manifest.json"))
	return {"scope":"Actual settled private idle is bent-knee skating posture, not standing stature or Main acceptance.",
		"native_head_local_bounds":head_mesh.mesh.get_aabb(),"native_head_world_y_m":[head_min,head_max],
		"actual_lowest_visible_sole_world_y_m":sole_min,"actual_idle_head_to_sole_height_m":head_max-sole_min,
		"actual_leg_lengths":lengths,"authored_arm_lengths_m":[.30,.29],"authored_shoulder_width_m":.43,
		"authored_hip_width_m":.21,"source_revision":source.source_revision,"source_ancestry_adult_identity":source.adaptation.adult_morph,
		"source_head_mesh_sha256":FileAccess.get_sha256("res://game/art/assets/makehuman_hm08/head.obj"),
		"actual_rendered_head_derivative_sha256":FileAccess.get_sha256("res://game/art/rider/fitted_head.obj"),
		"board_source_unchanged_sha256":FileAccess.get_sha256("res://game/art/assets/tam_skateboard/tam_skateboard.glb")}

func _native_inventory() -> Dictionary:
	var result := {"triangles":0,"surfaces":0,"mesh_instances":0,"textures":{}}
	_collect_resources(rider,result)
	_collect_resources(board,result)
	var details := []
	var conservative_bytes := 0
	for texture: Texture2D in result.textures.values():
		var bytes := int(texture.get_width()*texture.get_height()*4.0*4.0/3.0)
		conservative_bytes += bytes
		details.append({"path":texture.resource_path,"width":texture.get_width(),"height":texture.get_height(),"conservative_rgba_mips_bytes":bytes})
	result.textures = details
	result.conservative_rgba_mips_bytes = conservative_bytes
	result.scope = "Actual instantiated private Rider and native Board. Source triangle count, all active material Texture2D properties; conservative uncompressed RGBA mip estimate, not target GPU allocation or timing."
	if result.triangles>65000 or conservative_bytes>64*1024*1024:
		failures += 1
		push_error("Private fitted native resource budget exceeded.")
	return result

func _collect_resources(node: Node,result: Dictionary) -> void:
	if node is MeshInstance3D and node.mesh != null:
		result.mesh_instances += 1
		for surface in node.mesh.get_surface_count():
			result.surfaces += 1
			var arrays: Array = node.mesh.surface_get_arrays(surface)
			var indices: Variant = arrays[Mesh.ARRAY_INDEX]
			result.triangles += indices.size()/3 if indices is PackedInt32Array and not indices.is_empty() else arrays[Mesh.ARRAY_VERTEX].size()/3
			var material: Material = node.get_active_material(surface)
			if material != null:
				for property: Dictionary in material.get_property_list():
					var value: Variant = material.get(property.name)
					if value is Texture2D:result.textures[value.get_instance_id()] = value
	for child in node.get_children():_collect_resources(child,result)
