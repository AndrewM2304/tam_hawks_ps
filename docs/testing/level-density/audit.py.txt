# Independent scalar point-to-segment/arc checks, spaced across the exported grid.
segments=[(np.array(a),np.array(b)) for r in data['rails'] for a,b in zip(r['points'],r['points'][1:])]
def minimum_distance(p):
    candidates=[]
    for a,b in segments:
        v=b-a; t=max(0,min(1,float(np.dot(p-a,v)/np.dot(v,v))))
        candidates.append(float(np.linalg.norm(p-a-t*v)))
    for face in data['transitions']:
        frame=face['transform']; local=np.dot(p-frame['origin'], np.array(frame['basis']).T)
        r=face['radius']; theta=max(0,min(face['angle'],float(np.arctan2(-local[2],r-local[1]))))
        q=np.array([max(-face['width']/2,min(face['width']/2,local[0])),r*(1-np.cos(theta)),-r*np.sin(theta)])
        candidates.append(float(np.linalg.norm(local-q)))
        for end in (0, face['angle']):
            endpoint=np.array([q[0],r*(1-np.cos(end)),-r*np.sin(end)])
            candidates.append(float(np.linalg.norm(local-endpoint)))
    return min(candidates)
sample_distances=[minimum_distance(np.array(data['standing_grid'][i]['at'])) for i in range(0,6976,173)]
