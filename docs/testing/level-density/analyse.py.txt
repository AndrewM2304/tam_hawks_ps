d = json.loads((LOCAL / 'live-layout.json').read_text())
pts = np.array([c['at'] for c in d['standing_grid']], dtype=float)
normals = np.array([c['normal'] for c in d['standing_grid']], dtype=float)
slopes = np.degrees(np.arccos(np.clip(normals[:, 1], -1, 1)))
segments = np.array([(a, b) for rail in d['rails'] for a, b in zip(rail['points'], rail['points'][1:])], dtype=float)
a, v = segments[:, 0], segments[:, 1] - segments[:, 0]
vv = np.sum(v*v, axis=1)
assert np.all(vv > 0)
nearest_rail = np.zeros(len(pts))
for start in range(0, len(pts), 256):
    p = pts[start:start+256]
    t = np.clip(np.sum((p[:, None, :] - a) * v, axis=2) / vv, 0, 1)
    dist2 = np.sum((p[:, None, :] - a - t[:, :, None] * v)**2, axis=2)
    nearest_rail[start:start+len(p)] = np.sqrt(np.min(dist2, axis=1))
# Exact closest point to the authored cylindrical arc strip, including its ends.
nearest_transition = np.full(len(pts), np.inf)
for face in d['transitions']:
    frame = face['transform']
    basis = np.array(frame['basis']).T
    local = (pts - frame['origin']) @ basis
    theta = np.clip(np.arctan2(-local[:, 2], face['radius'] - local[:, 1]), 0, face['angle'])
    q = np.column_stack((np.clip(local[:, 0], -face['width']/2, face['width']/2),
                         face['radius']*(1-np.cos(theta)), -face['radius']*np.sin(theta)))
    face_distance = np.linalg.norm(local-q, axis=1)
    # atan2 wraps below -pi/2 for some points above/behind the toe. The
    # closest restricted-arc endpoint can then be the lip rather than toe.
    for end in (0, face['angle']):
        endpoint = np.column_stack((q[:, 0], np.full(len(pts), face['radius']*(1-np.cos(end))),
                                    np.full(len(pts), -face['radius']*np.sin(end))))
        face_distance = np.minimum(face_distance, np.linalg.norm(local-endpoint, axis=1))
    nearest_transition = np.minimum(nearest_transition, face_distance)
nearest = np.minimum(nearest_rail, nearest_transition)

def summarize(mask):
    n = int(np.sum(mask))
    return {'clear_standing_samples': n,
            'nearest_grind_or_transition_median_m': round(float(np.median(nearest[mask])), 2),
            'nearest_grind_or_transition_p90_m': round(float(np.percentile(nearest[mask], 90)), 2),
            'within_8m_samples': int(np.sum(mask & (nearest <= 8))),
            'beyond_15m_samples': int(np.sum(mask & (nearest > 15))),
            'beyond_15m_and_slope_under_2deg_samples': int(np.sum(mask & (nearest > 15) & (slopes < 2))),
            'beyond_15m_percent': round(100*float(np.sum(mask & (nearest > 15)))/n, 1)}

route_lengths = {r['id']: round(float(np.linalg.norm(np.diff(np.array(r['points']), axis=0), axis=1).sum()), 3) for r in d['routes']}
districts = sorted(set(c['district'] for c in d['standing_grid']))
district_summary = {name: summarize(np.array([c['district'] == name for c in d['standing_grid']])) for name in districts}
south = (pts[:, 0] >= -280) & (pts[:, 0] <= 280) & (pts[:, 2] >= 130) & (pts[:, 2] <= 220)
metrics = {

    'standing_grid': {'spacing_m': 4, 'x_centres': '-278 + 4*i, i=0..139', 'z_centres': '-224 + 4*j, j=0..111',
                      'capsule_radius_m': .28, 'capsule_height_m': 1.7, 'sole_clearance_m': .06,

No line-of-sight, connected approach or trick chain is proven. Banks/stairs/manual space/gaps/NPCs and visual interest are not scored.',
    'counts': {'grind_tracks': len(d['rails']), 'grind_kinds': dict(Counter(r['kind'] for r in d['rails'])),
               'route_definitions': len(d['routes']), 'transition_faces': len(d['transitions']), 'transition_arenas': 2},
    'all_samples': summarize(np.full(len(pts), True)), 'by_district_label': district_summary,
    'south_landscape_samples': summarize(south),
    'source_dimensions_m': {'south_landscape': [560, 90], 'north_quay': [300, 23], 'south_quay': [300, 14],
                          'bridge_x_separation': 133},
    'route_polyline_lengths_m': route_lengths,
    'illustrative_travel_seconds': {k: {'at_constant_8mps': round(v/8, 2), 'at_constant_15mps': round(v/15, 2)}
                                   for k, v in route_lengths.items()},
    'travel_limit': 'Distance/speed illustrations only; no acceleration, cornering, grade or real input is simulated. The actual rendered traverse observations are separate in STRICT-DENSITY-REVIEW.',
}
