#!/usr/bin/env python3
import os,sys,time,json,signal
from pathlib import Path
log=Path(os.environ['FAKE_RECORD']);mode=os.environ.get('FAKE_MODE','success')
def record(kind):
 with log.open('a') as f:f.write(json.dumps({'kind':kind,'pid':os.getpid(),'ppid':os.getppid(),'args':sys.argv[1:]})+'\n')
if '--version' in sys.argv:
 print('4.6.3.stable.official.synthetic');sys.exit(0)
if '--import' not in sys.argv:
 record('game');print('FAKE GAME START');sys.exit(0)
record('import')
if mode=='script-error':print('SCRIPT ERROR: synthetic parse failure');sys.exit(0)
if mode=='exit-seven':print('synthetic native exit');sys.exit(7)
if mode=='fork-hang':
 child=os.fork()
 if child==0:
  record('worker');signal.signal(signal.SIGTERM,signal.SIG_IGN)
  while True:time.sleep(.1)
if mode in ['hang','fork-hang']:
 print('import started',flush=True)
 while True:time.sleep(.1)
print('native progress line',flush=True);time.sleep(.03);print('last unterminated line',end='',flush=True)
