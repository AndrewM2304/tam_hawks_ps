if '--version' in sys.argv:print('4.6.3.stable.official.synthetic');sys.exit(0)
if '--import' not in sys.argv:record('game');print('FAKE GAME START');sys.exit(0)
record('import')
if mode.startswith('fork'):
 child=os.fork()
 if child==0:
  record('worker');signal.signal(signal.SIGTERM,signal.SIG_IGN)
  while True:time.sleep(.02)
 time.sleep(.05)
if mode in ['hang','fork-hang']:
 print('import started',flush=True)
 while True:time.sleep(.02)
print('native progress line',flush=True);print('last unterminated line',end='',flush=True)
