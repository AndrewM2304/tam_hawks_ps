extends SceneTree
const Main=preload("res://game/main.gd")
const Cable=preload("res://.local/cable_east_driver.gd")
const GapObserver=preload("res://.local/gap_observer.gd")
const ScoreObserver=preload("res://.local/score_observer.gd")
var passes:=0
var failures:Array[String]=[]
var samples:Array=[]
var game:Node
var gap_probe:RefCounted
var score_probe:RefCounted
func _initialize()->void:run.call_deferred()
func check(ok:bool,label:String)->void:
 if ok:passes+=1
 else:failures.append(label)
 print("PASS "if ok else"FAIL ",label)
func observe(stage:String,g:Node,tick:int)->void:
 samples.append({"run":gap_probe.run_label,"tick":tick,"stage":stage,"physics":Engine.get_physics_frames(),"process":Engine.get_process_frames(),"position":[g.skater.global_position.x,g.skater.global_position.y,g.skater.global_position.z],"airborne":g.skater.airborne,"grinding":g.skater.grinding,"grind_id":str(g.skater._grind.line.id)if g.skater.grinding else"","floor":g.skater.is_on_floor(),"total":g.score.total_score,"combo":g.score.get_combo_value(),"combo_base":g.score.combo_base,"combo_multiplier":g.score.combo_multiplier,"last_identity":g.score._last_identity,"history":g.score.trick_history.duplicate(),"last_gap_call":gap_probe.rows.size()-1})
func encode(value:Variant)->Variant:
 if value is Vector3:return [value.x,value.y,value.z]
 if value is Array:
  var result:Array=[]
  for v:Variant in value:result.append(encode(v))
  return result
 if value is Dictionary:
  var result:Dictionary={}
  for k:Variant in value:result[str(k)]=encode(value[k])
  return result
 return value
func run()->void:
 game=Main.new()
 check(not game.persistence_enabled,"script launch disables profile persistence before tree entry")
 if game.persistence_enabled:game.free();finish();return
 gap_probe=GapObserver.new();gap_probe.target=game
 score_probe=ScoreObserver.new()
 game.gaps=gap_probe
 game.score=score_probe
 root.add_child(game)
 check(not game.music._can_control and not game.music._persist,"actual default Main isolates personal Music/profile")
 var gaps:Array=encode(game.world.get_gap_definitions())
 for entry:Dictionary in [{"run":"east_southbound","direction":1.0,"wait":0},{"run":"east_northbound","direction":-1.0,"wait":0},{"run":"east_south_wait_12","direction":1.0,"wait":12}]:
  Cable.extra_cable_wait=int(entry.wait)
  gap_probe.run_label=str(entry.run)
  score_probe.run_label=gap_probe.run_label
  await Cable._drive_player(self,game,check,float(entry.direction),observe)
 check(gap_probe.mismatches.is_empty(),"super-delegating tracker returns exact original awards/state every actual update")
 var data:Dictionary={"passed":passes,"failed":failures.size(),"failures":failures,"definitions":gaps,"gap_updates":gap_probe.rows,"score_events":score_probe.events,"samples":samples,"shadow_mismatches":gap_probe.mismatches,"scope":"Derived EAST mirrored driver: original WEST functions use mirrored initial X, canonical side IDs and mapped steering sign; original timing for both directions plus SOUTH wait12 ordinary held-grind frames. No per-tick state or geometry writes. Gap/score observers delegate to original methods; shadow tracker results never feed game scoring. No extra spatial queries/draws or per-tick physical state writes; no performance/render/M5 claim."}
 var out:=FileAccess.open(OS.get_environment("TAM_GAP_COMBO_OUTPUT"),FileAccess.WRITE)
 if out==null:push_error("Cannot write gap/combo native evidence");quit(2);return
 out.store_string(JSON.stringify(data,"\t",true,true)+"\n");out.close()
 game.queue_free();await process_frame
 print("GAP_COMBO_NATIVE ",passes," passed, ",failures.size()," failed")
 quit(0 if failures.is_empty()else 1)
func finish()->void:
 print("GAP_COMBO_NATIVE ",passes," passed, ",failures.size()," failed")
 quit(1)
