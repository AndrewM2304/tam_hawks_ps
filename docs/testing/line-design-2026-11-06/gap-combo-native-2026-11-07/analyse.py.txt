def segdist(a,b,p):
 v=[b[i]-a[i]for i in range(3)];n=sum(x*x for x in v);t=max(0,min(1,sum((p[i]-a[i])*v[i]for i in range(3))/n))if n else 0
 return dist([a[i]+t*v[i]for i in range(3)],p),t
