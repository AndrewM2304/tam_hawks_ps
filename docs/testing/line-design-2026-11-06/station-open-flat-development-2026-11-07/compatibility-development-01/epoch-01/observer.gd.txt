func _protected_widths(game: Node) -> void:
	var packet: Array[Dictionary]=[]
	for item: Dictionary in [
		{"id":"original north low-bypass","from":Vector3(-82,2,34),"to":Vector3(-82,2,-34)},
		{"id":"original career diagonal","from":Vector3(-62,2,22),"to":Vector3(-82,2,34)},
		{"id":"new north-turn apron","from":Vector3(-75.4,2,42.3),"to":Vector3(-75.4,2,22)}]:
		var from: Vector3=item.from
		var to: Vector3=item.to
		var side:=Vector3((to-from).z,0,-(to-from).x).normalized()
		var clear:=true;var supported:=true;var failures: Array[Dictionary]=[]
		for column: int in 9:
			var offset: float=-2.0+float(column)*.5
			var a: Vector3=from+side*offset
			var b: Vector3=to+side*offset
			var sweep: bool=Routes._clear_sweep(game.world,a,b) and Routes._clear_sweep(game.world,b,a)
			clear=clear and sweep
			if not sweep:failures.append({"offset":offset,"from":a,"to":b,"kind":"two-way native capsule sweep"})
			var count:=maxi(2,ceili(a.distance_to(b)))
			for step: int in count+1:
				var point: Vector3=a.lerp(b,float(step)/count)
				var hit:=game.world.get_world_3d().direct_space_state.intersect_ray(PhysicsRayQueryParameters3D.create(point+Vector3.UP*.12,point-Vector3.UP*.12,1))
				var ok: bool=not hit.is_empty() and Vector3(hit.position).distance_to(point)<.002 and Vector3(hit.normal).dot(Vector3.UP)>.999
				supported=supported and ok
				if not ok and failures.size()<16:failures.append({"at":point,"kind":"fixed native floor support","hit":hit})
		_check(clear and supported,"station compatibility: supported four-metre body-centre band "+String(item.id))
		packet.append({"id":item.id,"width_m":4.0,"native_capsule_sweeps":18,"support_columns":9,"clear":clear,"supported":supported,"failures":failures})
	print("STATION_COMPAT_CORRIDORS ",JSON.stringify(packet))
