extends SceneTree
const Main=preload("res://game/main.gd")
const Tests=preload("res://tests/civic_edge_tests.gd")
var passed:=0
var failed:=0
func _initialize()->void:call_deferred("run")
func check(ok:bool,label:String)->void:
 if ok:passed+=1;print("PASS: ",label)
 else:failed+=1;print("FAIL: ",label)
func run()->void:
 var g:=Main.new();g.persistence_enabled=false;root.add_child(g);g.start_mode(true)
 for i in 30:await physics_frame;await process_frame
 await Tests.run(self,g,check)
 print("RESULT: %d passed, %d failed"%[passed,failed]);quit(1 if failed else 0)
