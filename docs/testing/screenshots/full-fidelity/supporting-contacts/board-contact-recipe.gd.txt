extends SceneTree
var game:Node
var checks:=0
var failures:=0
func _initialize()->void:_capture.call_deferred()
func _check(ok:bool,label:String)->void:
 checks+=1
 if not ok:failures+=1
 print(('PASS ' if ok else 'FAIL ')+label)
func _hashes()->Dictionary:
 var result={}
 for path:String in ['game/player/skater_visual.gd','game/player/skater.gd','game/player/skater_footwear.gd','game/player/skater_anatomy.gd','game/player/skater_garments.gd','game/tricks/trick_catalog.gd','project.godot','.local/board_contact_capture.gd']:
  result[path]=FileAccess.get_sha256('res://'+path)
 return result
func _frames(count:int)->void:
 for f in count:
  await physics_frame
  await process_frame
func _capture()->void:
 var before=_hashes()
 root.size=Vector2i(1280,800)
 DirAccess.make_dir_recursive_absolute('res://.local/board-contact-close')
 game=preload('res://game/main.gd').new()
 game.persistence_enabled=false
 root.add_child(game)
 await process_frame
 game.start_mode(true)
 await _frames(50)
 _check(game.skater.is_on_floor() and not game.skater.airborne and not game.skater.bail_active,'real Main native settled stance')
 game.skater.set_physics_process(false)
 game.skater.set_process(false)
 game.hud.visible=false
 var camera=Camera3D.new()
 camera.fov=44
 camera.near=.03
 root.add_child(camera)
 camera.current=true
 var center:Vector3=game.skater._board_visual.global_position
 var views=[{'id':'caught-shoes-and-flat-deck','offset':Vector3(.83,.46,-.68),'aim':Vector3(0,.10,0)}, {'id':'curved-kicks-and-flush-bolts','offset':Vector3(.67,.21,.51),'aim':Vector3(0,.045,0)}]
 var images=[]
 for view:Dictionary in views:
  camera.global_position=center+view.offset
  camera.look_at(center+view.aim)
  await process_frame
  RenderingServer.force_draw(true,1.0/60.0)
  var path='res://.local/board-contact-close/'+view.id+'.png'
  _check(root.get_texture().get_image().save_png(path)==OK,'captured actual native board inspection '+view.id)
  images.append({'id':view.id,'path':path,'sha256':FileAccess.get_sha256(path),'camera_offset':var_to_str(view.offset),'aim':var_to_str(view.aim)})
 var after=_hashes()
 _check(before==after,'all board/rig/controller/capture inputs unchanged')
 var manifest={'scope':'Actual native Main settled stance, inspection camera; detailed board/foot surface inspection, not a gameplay camera or target-Mac acceptance.','resolution':[1280,800],'images':images,'source_before':before,'source_after':after,'inputs_unchanged':before==after}
 var file=FileAccess.open('res://.local/board-contact-close/manifest.json',FileAccess.WRITE)
 file.store_string(JSON.stringify(manifest,'\t'));file.close()
 game.queue_free();await process_frame
 print('BOARD_CLOSE_CAPTURE ',checks-failures,' passed ',failures,' failed')
 quit(0 if failures==0 else 1)
