func _part_inside(part:MeshInstance3D,body:StaticBody3D)->Dictionary:
 var unique={}
 for point:Vector3 in part.mesh.get_faces():unique[point]=true
 var inside:=0
 var deepest:=0.0
 var example=Vector3.ZERO
 var finite_footprint:=0
 for point:Vector3 in unique:
  var world:Vector3=part.to_global(point)
  for child:Node in body.get_children():
   if not child is CollisionShape3D:continue
   var local:Vector3=child.to_local(world)
   var penetration:=0.0
   if child.shape is BoxShape3D:
    var half:Vector3=child.shape.size*.5
    if absf(local.x)<half.x-.00005 and absf(local.z)<half.z-.00005:
     finite_footprint+=1
     if local.y<half.y-.0005 and local.y> -half.y+.00005:penetration=half.y-local.y
   elif child.shape is CylinderShape3D:
    var radial=Vector2(local.x,local.z).length()
    var half_height:float=child.shape.height*.5
    var radius:float=child.shape.radius
    if absf(local.y)<half_height-.00005:
     var up:Vector3=(child.global_basis.inverse()*Vector3.UP).slide(Vector3.UP).normalized()
     var across:Vector3=Vector3.UP.cross(up).normalized()
     if absf(local.dot(across))<radius-.00005:finite_footprint+=1
     if radial<radius-.0005:penetration=minf(radius-radial,half_height-absf(local.y))
   if penetration>0:
    inside+=1
    if penetration>deepest:deepest=penetration;example=world
 var wheel:bool=part.mesh is CylinderMesh and is_equal_approx(part.mesh.top_radius,.032) and is_equal_approx(part.mesh.height,.032)
 return {'wheel':wheel,'part':str(part.name),'vertices':unique.size(),'inside':inside,'depth':deepest,'example':var_to_str(example),'finite_footprint':finite_footprint}
