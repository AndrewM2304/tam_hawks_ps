extends SceneTree
const Tests = preload("res://tests/skatepark_navigation_tests.gd")
var passed := 0
var failed := 0
func _initialize() -> void: _run.call_deferred()
func _run() -> void:
	await Tests.run_ui(self,_check)
	print("SKATEPARK NAVIGATION UI RESULT: %d passed, %d failed" % [passed,failed])
	quit(1 if failed else 0)
func _check(ok: bool, label: String) -> void:
	if ok: passed += 1
	else:
		failed += 1
		push_error("FAIL "+label)
	print(("PASS " if ok else "FAIL ")+label)
