extends SceneTree
const Main = preload("res://game/main.gd")
const Tests = preload("res://tests/skatepark_navigation_tests.gd")
var passed := 0
var failed := 0
func _initialize() -> void: _run.call_deferred()
func _run() -> void:
	var game = Main.new()
	game.persistence_enabled = false
	root.add_child(game)
	for frame in 35:
		await physics_frame
		await process_frame
	await Tests.run_world(self,game,_check)
	game.queue_free()
	await process_frame
	print("SKATEPARK NAVIGATION WORLD RESULT: %d passed, %d failed" % [passed,failed])
	quit(1 if failed else 0)
func _check(ok: bool, label: String) -> void:
	if ok: passed += 1
	else:
		failed += 1
		push_error("FAIL "+label)
	print(("PASS " if ok else "FAIL ")+label)
