extends SceneTree
const Tests = preload("res://tests/route_availability_tests.gd")
var passed := 0
var failed := 0
func _initialize() -> void: _run.call_deferred()
func _run() -> void:
	await Tests.run_ui(self,_check)
	if DisplayServer.get_name() != "headless": await Tests.run_rendered(self,_check)
	print("ROUTE AVAILABILITY RESULT: %d passed, %d failed" % [passed,failed])
	quit(1 if failed else 0)
func _check(ok: bool, message: String) -> void:
	if ok: passed += 1
	else:
		failed += 1
		push_error("FAIL " + message)
