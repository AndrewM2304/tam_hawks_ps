extends SceneTree
const Main = preload("res://game/main.gd")
var game: Node
func _initialize() -> void: _run.call_deferred()
func _run() -> void:
	if DisplayServer.get_name() == "headless":
		quit(1)
		return
	root.size = Vector2i(1280,960)
	root.content_scale_size = Vector2i(1280,960)
	game = Main.new()
	game.persistence_enabled = false
	root.add_child(game)
	for frame in 45:
		await physics_frame
		await process_frame
	game.hud.visible = false
	game.skater.set_physics_process(false)
	game.set_process(false)
	game.set_physics_process(false)
	var camera := Camera3D.new()
	camera.fov = 35
	camera.near = 0.05
	game.add_child(camera)
	camera.make_current()
	var actors: Array[Node] = []
	preload("res://tests/activity_actor_tests.gd")._find_actors(game,actors)
	DirAccess.make_dir_recursive_absolute("res://.local/activity-actors")
	for actor: Node3D in actors:
		camera.global_position = actor._body.global_position+Vector3(2.2,1.35,-3.8)
		camera.look_at(actor._body.global_position+Vector3.UP*0.90)
		for frame in 8: await process_frame
		await RenderingServer.frame_post_draw
		var image := root.get_texture().get_image()
		var path := "res://.local/activity-actors/"+str(actor.name)+".png"
		image.save_png(path)
		print("ACTIVITY CAPTURE actual Main ",actor.name," ",path," body=",actor._body.global_position," ground=",actor.get_ground_contacts())
	quit(0)
