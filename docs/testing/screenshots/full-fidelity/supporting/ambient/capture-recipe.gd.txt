extends SceneTree
var game: Node
var captures: Array[Dictionary] = []
var failures:=0
const OUTPUT="res://.local/ambient-adult-capture"
func _initialize(): call_deferred("_run")
func _run():
 if DisplayServer.get_name()=="headless":
  push_error("Ambient adult capture needs a rendering display")
  quit(1)
  return
 game=preload("res://game/main.gd").new()
 game.persistence_enabled=false
 root.add_child(game)
 game.start_mode(true)
 await _frames(35)
 game.set_physics_process(false)
 game.skater.set_physics_process(false)
 game.skater.set_process(false)
 game.hud.visible=false
 var camera:=Camera3D.new()
 game.add_child(camera)
 camera.fov=45
 camera.near=.07
 camera.current=true
 var actors=get_nodes_in_group("ambient_adults")
 var index:=0
 for actor in actors:
  index+=1
  var body:Node3D=actor.get_node("AdultActivityBody")
  camera.global_position=body.to_global(Vector3(2.2,1.7,-3.2))
  camera.look_at(body.to_global(Vector3(0,.95,0)))
  await _frames(2)
  RenderingServer.force_draw(true,1.0/60.0)
  await process_frame
  RenderingServer.force_draw(true,1.0/60.0)
  var label=["01-civic-adult.png","02-station-adult.png","03-barras-adult.png"][index-1]
  var image:=root.get_texture().get_image()
  var result=image.save_png(OUTPUT.path_join(label))
  if result!=OK: failures+=1
  captures.append({"file":label,"root":var_to_str(actor.global_position),"yaw":actor.rotation.y,"camera":var_to_str(camera.global_position),"body":var_to_str(body.global_position),"contacts":var_to_str(actor.get_ground_contacts()),"grounded":actor._grounded,"resolution":[image.get_width(),image.get_height()]})
 var manifest={"production_main":true,"profile_writes_enabled":false,"captures":captures,"failures":failures,"engine":Engine.get_version_info().string,"renderer":RenderingServer.get_current_rendering_method()}
 var file:=FileAccess.open(OUTPUT.path_join("manifest.json"),FileAccess.WRITE)
 file.store_string(JSON.stringify(manifest,"\t"))
 file.close()
 print("AMBIENT_CAPTURE_RESULT frames=%d failures=%d"%[captures.size(),failures])
 game.queue_free()
 await process_frame
 quit(1 if failures>0 or captures.size()!=3 else 0)
func _frames(count:int):
 for i in count:
  await physics_frame
  await process_frame
